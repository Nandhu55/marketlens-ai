    26→// ── Web Search ──────────────────────────────────────────────────
    27→export interface SearchResult {
    28→  url: string;
    29→  name: string;
    30→  snippet: string;
    31→  host_name: string;
    32→  rank: number;
    33→  date: string;
    34→  favicon: string;
    35→}
    36→
    37→export async function webSearch(
    38→  query: string,
    39→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    40→): Promise<SearchResult[]> {
    41→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    42→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    43→
    44→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    45→  if (cached) return cached;
    46→
    47→  const zai = await getZAI();
    48→  const args: Record<string, unknown> = { query, num };
    49→  if (recencyDays) args.recency_days = recencyDays;
    50→
    51→  try {
    52→    const results = await Promise.race([
    53→      zai.functions.invoke('web_search', args),
    54→      new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    55→    ]);
    56→    const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    57→    setCache(cacheKey, typed);
    58→    return typed;
    59→  } catch (err) {
    60→    throw err;
    61→  }
    62→}

    65→export async function readPage(url: string, cacheTtlMs = 600_000): Promise<{
    66→  title: string;
    67→  html: string;
    68→  url: string;
    69→  publishedTime?: string;
    70→}> {
    71→  const cacheKey = `page:${url}`;
    72→  const cached = getCached<{ title: string; html: string; url: string; publishedTime?: string }>(cacheKey, cacheTtlMs);
    73→  if (cached) return cached;
    74→
    75→  const zai = await getZAI();
    76→  const result = await Promise.race([
    77→    zai.functions.invoke('page_reader', { url }),
    78→    new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    79→  ]);
    80→
    81→  const data = {
    82→    title: result.data?.title || '',
    83→    html: result.data?.html || '',

    85→    publishedTime: result.data?.publishedTime || result.data?.publish_time,
    86→  };
    87→
    88→  setCache(cacheKey, data);
    89→  return data;
    90→}

    92→// ── LLM Chat ────────────────────────────────────────────────────
    93→export async function llmChat(
    94→  systemPrompt: string,
    95→  userMessage: string,
    96→  options?: { cacheTtlMs?: number; cacheKey?: string }
    97→): Promise<string> {
    98→  const { cacheTtlMs = 600_000, cacheKey } = options || {};
    99→
   100→  if (cacheKey) {
   101→    const cached = getCached<string>(cacheKey, cacheTtlMs);
   102→    if (cached) return cached;
   103→  }
   104→
   105→  const zai = await getZAI();
   106→  const completion = await Promise.race([
   107→    zai.chat.completions.create({
   108→      messages: [
   109→        { role: 'assistant', content: systemPrompt },
   110→        { role: 'user', content: userMessage },
   111→      ],
   112→      thinking: { type: 'disabled' },
   113→    }),
   114→    new Promise((_, reject) => setTimeout(() => reject(new Error('LLM Timeout')), 20000)),
   115→  ]) as Awaited<ReturnType<typeof zai.chat.completions.create>>;
   116→
   117→
   118→  const response = completion.choices[0]?.message?.content || '';
   119→  if (cacheKey) setCache(cacheKey, response);
   120→  return response;
   121→}

   137→// ── Helpers ─────────────────────────────────────────────────────
   138→function toNum(val: unknown, fallback = 0): number {
   139→  if (typeof val === 'number') return val;
   140→  if (typeof val === 'string') {
   141→    const parsed = parseFloat(val.replace(/[^0-9.\-]/g, ''));
   142→    return isNaN(parsed) ? fallback : parsed;
   143→  }
   144→  return fallback;
   145→}

   147→function clamp(val: number, min: number, max: number): number {
   148→  return Math.min(max, Math.max(min, val));
   149→}

   151→function parseJSON<T>(text: string): T | null {
   152→  try {
   153→    const jsonStr = text.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
   154→    return JSON.parse(jsonStr) as T;
   155→  } catch {
   156→    return null;
   157→  }
   158→}

   335→// ── Real News Fetching + LLM Sentiment ──────────────────────────
   336→export interface RealNewsArticle {
   337→  title: string;
   338→  url: string;
   339→  source: string;
   340→  snippet: string;
   341→  date: string;
   342→  sentiment: 'positive' | 'negative' | 'neutral';
   343→  sentimentScore: number;
   344→  positivePct: number;
   345→  negativePct: number;
   346→  neutralPct: number;
   347→}
   348→
   349→export async function fetchRealNews(symbol: string): Promise<{
   350→  articles: RealNewsArticle[];
   351→  summary: {
   352→    overallScore: number;
   353→    positivePct: number;
   354→    negativePct: number;
   355→    neutralPct: number;
   356→    positiveCount: number;
   357→    negativeCount: number;
   358→    neutralCount: number;
   359→    trending: string[];
   360→  };
   361→} | null> {
   362→  const cacheKey = `news:${symbol}`;
   363→  const cached = getCached<ReturnType<typeof fetchRealNews>>(cacheKey, 600_000);
   364→  if (cached) return cached;
   365→
   366→  try {
   367→    const companyName = getCompanyName(symbol);
   368→    const query = `${symbol} ${companyName} stock news today latest`;
   369→    const results = await webSearch(query, { num: 10, recencyDays: 7 });
   370→
   371→    if (!results.length) return null;
   372→
   373→    const articlesText = results.slice(0, 10)
   374→      .map((r, i) => `[${i + 1}] Title: ${r.name}\nSource: ${r.host_name}\nSnippet: ${r.snippet}\nDate: ${r.date || 'Recent'}`)
   375→      .join('\n\n');
   376→
   377→    const systemPrompt = `You are a financial sentiment analysis AI (FinBERT-class). Analyze the sentiment of each news article about ${symbol} stock.
   378→
   379→For EACH article, determine:
   380→- sentiment: "positive", "negative", or "neutral"
   381→- score: 0-100 (where >60 = positive, <40 = negative, 40-60 = neutral)
   382→- positivePct: confidence for positive sentiment (0-100)
   383→- negativePct: confidence for negative sentiment (0-100)
   384→- neutralPct: confidence for neutral sentiment (0-100)
   385→
   386→Also identify 4-6 trending topics/themes across all articles.
   387→
   388→Return ONLY a valid JSON object:
   389→{
   390→  "articles": [{ "index": 1, "sentiment": "positive", "score": 72, "positivePct": 65, "negativePct": 15, "neutralPct": 20 }],
   391→  "trending": ["Topic 1", "Topic 2"]
   392→}
   393→
   394→Analyze ALL ${Math.min(results.length, 10)} articles. Respond with JSON only.`;
   395→
   396→    const llmResponse = await llmChat(systemPrompt, `Analyze sentiment for these news articles about ${symbol}:\n\n${articlesText}`);
   397→
   398→    const parsed = parseJSON<{
   399→      articles: Array<{ index: number; sentiment: string; score: number; positivePct: number; negativePct: number; neutralPct: number }>;
   400→      trending: string[];
   401→    }>(llmResponse);
   402→
   403→    if (!parsed) return null;
   404→
   405→    const articles: RealNewsArticle[] = results.slice(0, 10).map((r, i) => {
   406→      const analysis = parsed.articles?.find((a) => a.index === i + 1);
   407→      return {
   408→        title: r.name,
   409→        url: r.url,
   410→        source: r.host_name,
   411→        snippet: r.snippet,
   412→        date: r.date || 'Recent',
   413→        sentiment: (analysis?.sentiment || 'neutral') as 'positive' | 'negative' | 'neutral',
   414→        sentimentScore: analysis?.score ?? 50,
   415→        positivePct: analysis?.positivePct ?? 33,
   416→        negativePct: analysis?.negativePct ?? 33,
   417→        neutralPct: analysis?.neutralPct ?? 34,
   418→      };
   419→    });
   420→
   421→    const positiveCount = articles.filter((a) => a.sentiment === 'positive').length;
   422→    const negativeCount = articles.filter((a) => a.sentiment === 'negative').length;
   423→    const neutralCount = articles.filter((a) => a.sentiment === 'neutral').length;
   424→    const avgPositive = articles.reduce((s, a) => s + a.positivePct, 0) / articles.length;
   425→    const avgNegative = articles.reduce((s, a) => s + a.negativePct, 0) / articles.length;
   426→    const avgNeutral = 100 - avgPositive - avgNegative;
   427→
   428→    const result = {
   429→      articles,
   430→      summary: {
   431→        overallScore: Math.round(50 + (avgPositive - avgNegative) * 0.5),
   432→        positivePct: Math.round(avgPositive),
   433→        negativePct: Math.round(avgNegative),
   434→        neutralPct: Math.round(Math.max(avgNeutral, 0)),
   435→        positiveCount,
   436→        negativeCount,
   437→        neutralCount,
   438→        trending: parsed.trending || [],
   439→      },
   440→    };
   441→
   442→    setCache(cacheKey, result);
   443→    return result;
   444→  } catch (error) {
   445→    console.error(`Failed to fetch real news for ${symbol}:`, error);
   446→    return null;
   447→  }
   448→}

   556→    const vol = clamp(toNum(parsed.volatility, 2.5), 0.5, 8);
   557→
   558→    const result: RealPrediction = {
   559→      bullishProbability: Math.round(bullish),
   560→      bearishProbability: Math.round(100 - bullish),
   561→      volatility: parseFloat(vol.toFixed(2)),
   562→      predictedPrice: parseFloat(predPrice.toFixed(2)),
   563→      upperBound: parseFloat(clamp(toNum(parsed.upperBound, predPrice * 1.015), predPrice, predPrice * 1.05).toFixed(2)),
   564→      lowerBound: parseFloat(clamp(toNum(parsed.lowerBound, predPrice * 0.985), predPrice * 0.95, predPrice).toFixed(2)),
   565→      confidence: Math.round(clamp(toNum(parsed.confidence, 65), 40, 85)),
   566→      riskLevel: vol > 4 ? 'High' : vol > 2.5 ? 'Medium' : 'Low',
   567→      recommendation: (parsed.recommendation as string) || (bullish > 55 ? 'BUY' : bullish < 45 ? 'SELL' : 'HOLD'),
   568→      analysis: (parsed.analysis as string) || '',
   569→      keyFactors: Array.isArray(parsed.keyFactors) ? parsed.keyFactors.map(String) : [],
   570→      supportLevels: Array.isArray(parsed.supportLevels) ? parsed.supportLevels.map(Number).filter((n) => n > 0) : [],
   571→      resistanceLevels: Array.isArray(parsed.resistanceLevels) ? parsed.resistanceLevels.map(Number).filter((n) => n > 0) : [],
   572→    };
   573→
   574→    setCache(cacheKey, result);
   575→    return result;
   576→  } catch (error) {
   577→    console.error(`Failed to fetch prediction for ${symbol}:`, error);
   578→    return null;
   579→  }
   580→}

   703→// ── LLM Forecast (real, no fake history) ─────────────────────────
   704→export async function fetchRealForecast(
   705→  symbol: string,
   706→  currentPrice: number,
   707→  timeframe: string,
   708→  indicators: Record<string, number | undefined>
   709→): Promise<{
   710→  points: Array<{ date: string; predicted: number; lower: number; upper: number; confidence: number }>;
   711→  summary: {
   712→    predictedPrice: number;
   713→    expectedChange: number;
   714→    volatility: number;
   715→    confidence: number;
   716→    analysis: string;
   717→  };
   718→} | null> {
   719→  const cacheKey = `forecast:${symbol}:${timeframe}:${currentPrice.toFixed(2)}`;
   720→  const cached = getCached<ReturnType<typeof fetchRealForecast>>(cacheKey, 900_000);
   721→  if (cached) return cached;
   722→
   723→  try {
   724→    const companyName = getCompanyName(symbol);
   725→    const days = timeframe === 'tomorrow' ? 1 : timeframe === '3days' ? 3 : timeframe === '7days' ? 7 : 30;
   726→
   727→    const systemPrompt = `You are an AI stock forecaster. Generate a realistic price forecast for ${symbol} (${companyName}).
   728→
   729→Current data:
   730→- Price: $${currentPrice.toFixed(2)}
   731→- RSI: ${indicators.rsi?.toFixed(1) || 'N/A'}
   732→- MACD: ${indicators.macd?.toFixed(2) || 'N/A'}
   733→- SMA(20): $${indicators.sma20?.toFixed(2) || 'N/A'}
   734→- SMA(50): $${indicators.sma50?.toFixed(2) || 'N/A'}
   735→- ATR: ${indicators.atr?.toFixed(2) || 'N/A'}
   736→
   737→Timeframe: Next ${days} trading day(s).
   738→
   739→Return ONLY valid JSON:
   740→{
   741→  "points": [
   742→    {"date": "YYYY-MM-DD", "predicted": <number>, "lower": <number>, "upper": <number>, "confidence": <number 40-90>}
   743→  ],
   744→  "summary": {
   745→    "predictedPrice": <number>,
   746→    "expectedChange": <percent number>,
   747→    "volatility": <number>,
   748→    "confidence": <number>,
   749→    "analysis": "<2-3 sentence analysis>"
   750→  }
   751→}
   752→
   753→Rules:
   754→- Generate exactly ${days} forecast point(s)
   755→- Predicted price must be within 5% of current price for short timeframes
   756→- Confidence intervals should widen with time
   757→- Use realistic volatility based on ATR
   758→- Dates must be future trading days (skip weekends)
   759→Respond with JSON only.`;
   760→
   761→    const llmResponse = await llmChat(systemPrompt, `Generate ${days}-day forecast for ${symbol}.`);
   762→    const parsed = parseJSON<Record<string, unknown>>(llmResponse);
   763→    if (!parsed) return null;
   764→
   765→    const points = Array.isArray(parsed.points)
   766→      ? (parsed.points as Array<Record<string, unknown>>).map((p) => ({
   767→          date: String(p.date || ''),
   768→          predicted: clamp(toNum(p.predicted, currentPrice), currentPrice * 0.9, currentPrice * 1.1),
   769→          lower: clamp(toNum(p.lower, currentPrice), currentPrice * 0.9, currentPrice),
   770→          upper: clamp(toNum(p.upper, currentPrice), currentPrice, currentPrice * 1.1),
   771→          confidence: clamp(toNum(p.confidence, 60), 30, 90),
   772→        }))
   773→      : [];
   774→
   775→    const summary = parsed.summary as Record<string, unknown> | undefined;
   776→    const predictedPrice = points.length > 0 ? points[points.length - 1].predicted : currentPrice;
   777→    const expectedChange = ((predictedPrice - currentPrice) / currentPrice) * 100;
   778→
   779→    const result = {
   780→      points,
   781→      summary: {
   782→        predictedPrice,
   783→        expectedChange: parseFloat(expectedChange.toFixed(2)),
   784→        volatility: toNum(summary?.volatility, 2),
   785→        confidence: clamp(toNum(summary?.confidence, 60), 30, 90),
   786→        analysis: String(summary?.analysis || 'No analysis available.'),
   787→      },
   788→    };
   789→
   790→    setCache(cacheKey, result);
   791→    return result;
   792→  } catch (error) {
   793→    console.error(`Failed to fetch forecast for ${symbol}:`, error);
   794→    return null;
   795→  }
