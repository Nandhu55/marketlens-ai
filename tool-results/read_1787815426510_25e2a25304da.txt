   104→// ─── Chart Tooltip ───────────────────────────────────────────────────
   105→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color?: string }>; label?: string }) {
   106→  if (!active || !payload?.length) return null;
   107→  return (
   108→    <div className="market-tooltip px-3 py-2 text-xs">
   109→      <p className="text-slate-400 mb-1">{label}</p>
   110→      {payload.map((entry, i) => (
   111→        <p key={i} style={{ color: entry.color || '#fff' }} className="font-semibold">
   112→          {entry.dataKey}: {typeof entry.value === 'number' ? entry.value.toFixed(2) : entry.value}
   113→        </p>
   114→      ))}
   115→    </div>
   116→  );
   117→}

   127→// ─── Main LiveMarketPage ────────────────────────────────────────────
   128→export function LiveMarketPage() {
   129→  const { selectedStock, setSelectedStock } = useAppStore();
   130→  const [searchQuery, setSearchQuery] = useState('');
   131→  const [searchResults, setSearchResults] = useState<SearchResult[]>([]);
   132→  const [showSearch, setShowSearch] = useState(false);
   133→  const [activeTab, setActiveTab] = useState('price');
   134→  const [activeTimeframe, setActiveTimeframe] = useState(90);
   135→  const [stockData, setStockData] = useState<StockResponse | null>(null);
   136→  const [loading, setLoading] = useState(true);
   137→  const [searchLoading, setSearchLoading] = useState(false);
   138→  const searchRef = useRef<HTMLDivElement>(null);
   139→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   140→
   141→  // Fetch stock data
   142→  const fetchStockData = useCallback(async (symbol: string, days: number) => {
   143→    setLoading(true);
   144→    try {
   145→      const res = await fetch(`/api/stock?symbol=${symbol}&days=${days}`);
   146→      const data: StockResponse = await res.json();
   147→      setStockData(data);
   148→    } catch {
   149→      setStockData(null);
   150→    } finally {
   151→      setLoading(false);
   152→    }
   153→  }, []);
   154→
   155→  // Initial fetch
   156→  useEffect(() => {
   157→    fetchStockData(selectedStock.symbol, activeTimeframe);
   158→  }, [selectedStock.symbol, activeTimeframe, fetchStockData]);
   159→
   160→  // Search with debounce
   161→  useEffect(() => {
   162→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current);
   163→    if (searchQuery.length < 1) {
   164→      setSearchResults([]);
   165→      return;
   166→    }
   167→    setSearchLoading(true);
   168→    searchTimeoutRef.current = setTimeout(async () => {
   169→      try {
   170→        const res = await fetch(`/api/search?q=${encodeURIComponent(searchQuery)}`);
   171→        const data = await res.json();
   172→        setSearchResults(data.results || []);
   173→      } catch {
   174→        setSearchResults([]);
   175→      } finally {
   176→        setSearchLoading(false);
   177→      }
   178→    }, 300);
   179→    return () => {
   180→      if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current);
   181→    };
   182→  }, [searchQuery]);
   183→
   184→  // Close search on outside click
   185→  useEffect(() => {
   186→    const handleClick = (e: MouseEvent) => {
   187→      if (searchRef.current && !searchRef.current.contains(e.target as Node)) {
   188→        setShowSearch(false);
   189→      }
   190→    };
   191→    document.addEventListener('mousedown', handleClick);
   192→    return () => document.removeEventListener('mousedown', handleClick);
   193→  }, []);
   194→
   195→  const handleSelectStock = (result: SearchResult) => {
   196→    setSelectedStock({
   197→      symbol: result.symbol,
   198→      name: result.name,
   199→      price: result.price || 0,
   200→      change: result.change || 0,
   201→      changePercent: result.changePercent || 0,
   202→      volume: 0,
   203→      marketCap: 0,
   204→    });
   205→    setShowSearch(false);
   206→    setSearchQuery('');
   207→    setSearchResults([]);
   208→  };
   209→
   210→  // Prepare chart data
   211→  const ohlcv = stockData?.ohlcv || [];
   212→  const hasOHLCV = stockData?.hasOHLCV && ohlcv.length > 0;
   213→
   214→  const priceData = ohlcv.map((d) => ({
   215→    date: d.date.slice(5),
   216→    open: d.open,
   217→    high: d.high,
   218→    low: d.low,
   219→    close: d.close,
   220→    volume: d.volume,
   221→  }));
   222→
   223→  // Volume chart data
   224→  const volumeData = ohlcv.map((d) => ({
   225→    date: d.date.slice(5),
   226→    volume: d.volume,
   227→    close: d.close,
   228→    prevClose: d.open,
   229→  }));
   230→
   231→  // Indicator chart data from API series
   232→  const series = stockData?.series || null;
   233→  const indicatorData = series
   234→    ? ohlcv.map((d, i) => {
   235→        const point: Record<string, string | number> = { date: d.date.slice(5), close: d.close };
   236→        if (series.sma20?.[i] !== undefined && !isNaN(series.sma20[i])) point.sma20 = series.sma20[i];
   237→        if (series.sma50?.[i] !== undefined && !isNaN(series.sma50[i])) point.sma50 = series.sma50[i];
   238→        if (series.ema12?.[i] !== undefined && !isNaN(series.ema12[i])) point.ema12 = series.ema12[i];
   239→        if (series.rsi?.[i] !== undefined && !isNaN(series.rsi[i])) point.rsi = series.rsi[i];
   240→        if (series.bbUpper?.[i] !== undefined && !isNaN(series.bbUpper[i])) point.bbUpper = series.bbUpper[i];
   241→        if (series.bbMiddle?.[i] !== undefined && !isNaN(series.bbMiddle[i])) point.bbMiddle = series.bbMiddle[i];
   242→        if (series.bbLower?.[i] !== undefined && !isNaN(series.bbLower[i])) point.bbLower = series.bbLower[i];
   243→        // MACD series may be shorter — align from end
   244→        const macdLen = series.macd?.length || 0;
   245→        const macdIdx = i - (ohlcv.length - macdLen);
   246→        if (macdIdx >= 0 && series.macd?.[macdIdx] !== undefined && !isNaN(series.macd[macdIdx])) {
   247→          point.macd = series.macd[macdIdx];
   248→        }
   249→        const sigLen = series.macdSignal?.length || 0;
   250→        const sigIdx = i - (ohlcv.length - sigLen);
   251→        if (sigIdx >= 0 && series.macdSignal?.[sigIdx] !== undefined && !isNaN(series.macdSignal[sigIdx])) {
   252→          point.macdSignal = series.macdSignal[sigIdx];
   253→        }
   254→        const histLen = series.macdHistogram?.length || 0;
   255→        const histIdx = i - (ohlcv.length - histLen);
   256→        if (histIdx >= 0 && series.macdHistogram?.[histIdx] !== undefined && !isNaN(series.macdHistogram[histIdx])) {
   257→          point.macdHistogram = series.macdHistogram[histIdx];
   258→        }
   259→        return point;
   260→      })
   261→    : [];
   262→
   263→  const indicators = stockData?.indicators || null;
   264→
   265→  return (
   266→    <motion.div
   267→      variants={containerVariants}
   268→      initial="hidden"
   269→      animate="show"
   270→      className="space-y-6"
   271→    >
   272→      {/* ── Search Bar ────────────────────────────────────────── */}
   273→      <motion.div variants={itemVariants} ref={searchRef} className="relative">
   274→        <div className="relative">
   275→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   276→          <Input
   277→            value={searchQuery}
   278→            onChange={(e) => {
   279→              setSearchQuery(e.target.value);
   280→              setShowSearch(true);
   281→            }}
   282→            onFocus={() => setShowSearch(true)}
   283→            placeholder="Search stocks by name or symbol..."
   284→            className="glass-card pl-11 pr-10 py-3 h-12 text-sm bg-transparent border-white/10 focus:border-indigo-500/50 rounded-xl"
   285→          />
   286→          {searchQuery && (
   287→            <button
   288→              onClick={() => { setSearchQuery(''); setSearchResults([]); }}
   289→              className="absolute right-4 top-1/2 -translate-y-1/2 text-slate-500 hover:text-white transition-colors"
   290→            >
   291→              <X className="w-4 h-4" />
   292→            </button>
   293→          )}
   294→        </div>
   295→
   296→        {/* Search Dropdown */}
   297→        <AnimatePresence>
   298→          {showSearch && searchQuery.length >= 1 && (
   299→            <motion.div
   300→              initial={{ opacity: 0, y: -4 }}
   301→              animate={{ opacity: 1, y: 0 }}
   302→              exit={{ opacity: 0, y: -4 }}
   303→              transition={{ duration: 0.15 }}
   304→              className="absolute top-full left-0 right-0 mt-2 glass-card p-2 z-50 max-h-72 overflow-y-auto"
   305→            >
   306→              {searchLoading ? (
   307→                <div className="flex items-center justify-center py-6">
   308→                  <LoadingSpinner size={24} label="Searching..." />
   309→                </div>
   310→              ) : searchResults.length === 0 ? (
   311→                <div className="text-center py-6 text-slate-500 text-sm">No results found</div>
   312→              ) : (
   313→                searchResults.map((r) => (
   314→                  <button
   315→                    key={r.symbol}
   316→                    onClick={() => handleSelectStock(r)}
   317→                    className="w-full flex items-center justify-between px-3 py-2.5 rounded-xl hover:bg-white/5 transition-colors text-left"
   318→                  >
   319→                    <div>
   320→                      <div className="flex items-center gap-2">
   321→                        <span className="text-sm font-bold text-white">{r.symbol}</span>
   322→                        <Badge variant="outline" className="text-[9px] border-white/10 text-slate-400 px-1.5 py-0">
   323→                          {r.exchange}
   324→                        </Badge>
   325→                      </div>
   326→                      <p className="text-xs text-slate-400 mt-0.5">{r.name}</p>
   327→                    </div>
   328→                    <div className="text-right">
   329→                      {r.hasRealPrice ? (
   330→                        <>
   331→                          <p className="text-sm font-semibold text-white">${formatPrice(r.price)}</p>
   332→                          <p className={cn('text-xs font-medium', getChangeColor(r.changePercent))}>
   333→                            {formatPercent(r.changePercent)}
   334→                          </p>
   335→                        </>
   336→                      ) : (
   337→                        <p className="text-xs text-slate-500">No price</p>
   338→                      )}
   339→                    </div>
   340→                  </button>
   341→                ))
   342→              )}
   343→            </motion.div>
   344→          )}
   345→        </AnimatePresence>
   346→      </motion.div>
   347→
   348→      {/* ── Main Content ───────────────────────────────────────── */}
   349→      <div className="grid grid-cols-1 lg:grid-cols-4 gap-6">
   350→        {/* Chart Area */}
   351→        <motion.div variants={itemVariants} className="lg:col-span-3 space-y-4">
   352→          {/* Header with timeframe selector */}
   353→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   354→            <div>
   355→              <div className="flex items-center gap-3">
   356→                <h2 className="text-xl font-bold text-white">{selectedStock.symbol}</h2>
   357→                {stockData?.stock?.name && (
   358→                  <span className="text-sm text-slate-400">{stockData.stock.name}</span>
   359→                )}
   360→                {stockData?.realQuote && (
   361→                  <Badge className="bg-emerald-500/15 text-emerald-400 text-[10px] font-semibold border-0">
   362→                    LIVE
   363→                  </Badge>
   364→                )}
   365→              </div>
   366→              {stockData?.stock && (
   367→                <div className="flex items-center gap-3 mt-1">
   368→                  <span className="text-2xl font-bold text-white">${formatPrice(stockData.stock.price)}</span>
   369→                  <span className={cn('text-sm font-semibold flex items-center gap-1', getChangeColor(stockData.stock.changePercent))}>
   370→                    {stockData.stock.changePercent >= 0 ? <ArrowUpRight className="w-4 h-4" /> : <ArrowDownRight className="w-4 h-4" />}
   371→                    {formatPrice(stockData.stock.change)} ({formatPercent(stockData.stock.changePercent)})
   372→                  </span>
   373→                </div>
   374→              )}
   375→            </div>
   376→            <div className="flex items-center gap-2">
   377→              <div className="flex bg-white/5 rounded-xl p-1">
   378→                {TIMEFRAMES.map((tf) => (
   379→                  <button
   380→                    key={tf.days}
   381→                    onClick={() => setActiveTimeframe(tf.days)}
   382→                    className={cn(
   383→                      'px-3 py-1.5 text-xs font-medium rounded-lg transition-all',
   384→                      activeTimeframe === tf.days
   385→                        ? 'gradient-button text-white'
   386→                        : 'text-slate-400 hover:text-white hover:bg-white/5'
   387→                    )}
   388→                  >
   389→                    {tf.label}
   390→                  </button>
   391→                ))}
   392→              </div>
   393→            </div>
   394→          </div>
   395→
   396→          {/* Indicator Tabs + Chart */}
   397→          <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   398→            <TabsList className="bg-white/5 rounded-xl p-1 h-9">
   399→              <TabsTrigger value="price" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">Price</TabsTrigger>
   400→              <TabsTrigger value="sma" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">SMA/EMA</TabsTrigger>
   401→              <TabsTrigger value="rsi" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">RSI</TabsTrigger>
   402→              <TabsTrigger value="macd" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">MACD</TabsTrigger>
   403→              <TabsTrigger value="bb" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">Bollinger</TabsTrigger>
   404→            </TabsList>
   405→
   406→            <TabsContent value="price" className="mt-4">
   407→              {loading ? (
   408→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   409→                  <LoadingSpinner label="Loading price data..." />
   410→                </div>
   411→              ) : !hasOHLCV ? (
   412→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   413→                  <EmptyState
   414→                    icon={<Activity className="w-8 h-8" />}
   415→                    title="No Live Data Available"
   416→                    description={`No OHLCV data for ${selectedStock.symbol}. Try a different stock or refresh.`}
   417→                    action={
   418→                      <Button variant="outline" size="sm" onClick={() => fetchStockData(selectedStock.symbol, activeTimeframe)} className="gradient-button border-0">
   419→                        <RefreshCw className="w-3.5 h-3.5 mr-2" />
   420→                        Retry
   421→                      </Button>
   422→                    }
   423→                  />
   424→                </div>
   425→              ) : (
   426→                <>
   427→                  <div className="glass-card p-4">
   428→                    <ResponsiveContainer width="100%" height={300}>
   429→                      <ComposedChart data={priceData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   430→                        <defs>
   431→                          <linearGradient id="areaGrad" x1="0" y1="0" x2="0" y2="1">
   432→                            <stop offset="0%" stopColor="#6366f1" stopOpacity={0.15} />
   433→                            <stop offset="100%" stopColor="#6366f1" stopOpacity={0} />
   434→                          </linearGradient>
   435→                        </defs>
   436→                        <CartesianGrid strokeDasharray="3 3" stroke="rgba(148,163,184,0.06)" />
   437→                        <XAxis dataKey="date" axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} />
   438→                        <YAxis domain={['auto', 'auto']} axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} tickFormatter={(v: number) => `$${v.toFixed(0)}`} />
   439→                        <Tooltip content={<ChartTooltip />} />
   440→                        <Area type="monotone" dataKey="close" stroke="#6366f1" strokeWidth={2} fill="url(#areaGrad)" dot={false} />
   441→                        <Line type="monotone" dataKey="high" stroke="#10b981" strokeWidth={0.5} dot={false} strokeDasharray="2 2" />
   442→                        <Line type="monotone" dataKey="low" stroke="#ef4444" strokeWidth={0.5} dot={false} strokeDasharray="2 2" />
   443→                      </ComposedChart>
   444→                    </ResponsiveContainer>
   445→                  </div>
   446→                  {/* Volume bars */}
   447→                  <div className="glass-card p-4 mt-2">
   448→                    <div className="flex items-center gap-2 mb-2">
   449→                      <Volume2 className="w-3.5 h-3.5 text-slate-500" />
   450→                      <span className="text-xs text-slate-500 uppercase tracking-wider font-medium">Volume</span>
   451→                    </div>
   452→                    <ResponsiveContainer width="100%" height={80}>
   453→                      <BarChart data={volumeData} margin={{ top: 0, right: 5, left: -20, bottom: 0 }}>
   454→                        <XAxis dataKey="date" axisLine={false} tickLine={false} tick={false} />
   455→                        <YAxis axisLine={false} tickLine={false} tick={false} />
   456→                        <Tooltip content={<ChartTooltip />} />
   457→                        <Bar dataKey="volume" radius={[2, 2, 0, 0]}>
   458→                          {volumeData.map((entry, index) => (
   459→                            <Cell
   460→                              key={index}
   461→                              fill={entry.close >= entry.prevClose ? 'rgba(16,185,129,0.4)' : 'rgba(239,68,68,0.4)'}
   462→                            />
   463→                          ))}
   464→                        </Bar>
   465→                      </BarChart>
   466→                    </ResponsiveContainer>
   467→                  </div>
   468→                </>
   469→              )}
   470→            </TabsContent>
   471→
   472→            <TabsContent value="sma" className="mt-4">
   473→              {loading || !hasOHLCV || !series ? (
   474→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   475→                  {!hasOHLCV ? (
   476→                    <EmptyState icon={<Layers className="w-8 h-8" />} title="No Live Data Available" description="No OHLCV data to compute indicators" />
   477→                  ) : (
   478→                    <LoadingSpinner label="Loading indicators..." />
   479→                  )}
   480→                </div>
   481→              ) : (
   482→                <div className="glass-card p-4">
   483→                  <div className="flex items-center gap-4 mb-3">
   484→                    <div className="flex items-center gap-1.5"><div className="w-3 h-0.5 bg-indigo-400" /><span className="text-[10px] text-slate-400">SMA 20</span></div>
   485→                    <div className="flex items-center gap-1.5"><div className="w-3 h-0.5 bg-cyan-400" /><span className="text-[10px] text-slate-400">SMA 50</span></div>
   486→                    <div className="flex items-center gap-1.5"><div className="w-3 h-0.5 bg-amber-400" /><span className="text-[10px] text-slate-400">EMA 12</span></div>
   487→                  </div>
   488→                  <ResponsiveContainer width="100%" height={350}>
   489→                    <LineChart data={indicatorData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   490→                      <CartesianGrid strokeDasharray="3 3" stroke="rgba(148,163,184,0.06)" />
   491→                      <XAxis dataKey="date" axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} />
   492→                      <YAxis domain={['auto', 'auto']} axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} tickFormatter={(v: number) => `$${v.toFixed(0)}`} />
   493→                      <Tooltip content={<ChartTooltip />} />
