    72→// ─── Fear & Greed Semicircle Gauge ─────────────────────────────────────────
    73→function FearGreedGauge({ value }: { value: number }) {
    74→  const clamped = Math.max(0, Math.min(100, value));
    75→
    76→  const getZone = () => {
    77→    if (clamped <= 25) return { label: 'Extreme Fear', color: '#ef4444' };
    78→    if (clamped <= 45) return { label: 'Fear', color: '#f97316' };
    79→    if (clamped <= 55) return { label: 'Neutral', color: '#eab308' };
    80→    if (clamped <= 75) return { label: 'Greed', color: '#22c55e' };
    81→    return { label: 'Extreme Greed', color: '#10b981' };
    82→  };
    83→
    84→  const zone = getZone();
    85→
    86→  // Semicircle arc parameters
    87→  const width = 260;
    88→  const height = 145;
    89→  const cx = width / 2;
    90→  const cy = height - 10;
    91→  const radius = 100;
    92→  const strokeWidth = 18;
    93→
    94→  // Convert value (0-100) to angle (PI to 0, left to right)
    95→  const valueAngle = Math.PI - (clamped / 100) * Math.PI;
    96→  const startX = cx - radius;
    97→  const startY = cy;
    98→  const endX = cx + radius;
    99→  const endY = cy;
   100→
   101→  // Needle tip position
   102→  const needleLen = radius - strokeWidth - 5;
   103→  const needleTipX = cx - needleLen * Math.cos(valueAngle);
   104→  const needleTipY = cy - needleLen * Math.sin(valueAngle);
   105→
   106→  // Zone segments for background arc
   107→  const segments = [
   108→    { from: 0, to: 25, color: '#ef4444' },
   109→    { from: 25, to: 45, color: '#f97316' },
   110→    { from: 45, to: 55, color: '#eab308' },
   111→    { from: 55, to: 75, color: '#22c55e' },
   112→    { from: 75, to: 100, color: '#10b981' },
   113→  ];
   114→
   115→  const getArcPath = (fromPct: number, toPct: number) => {
   116→    const startAngle = Math.PI - (fromPct / 100) * Math.PI;
   117→    const endAngle = Math.PI - (toPct / 100) * Math.PI;
   118→    const sx = cx + radius * Math.cos(startAngle);
   119→    const sy = cy - radius * Math.sin(startAngle);
   120→    const ex = cx + radius * Math.cos(endAngle);
   121→    const ey = cy - radius * Math.sin(endAngle);
   122→    const largeArc = (fromPct - toPct) > 50 ? 1 : 0;
   123→    return `M ${sx} ${sy} A ${radius} ${radius} 0 ${largeArc} 0 ${ex} ${ey}`;
   124→  };
   125→
   126→  return (
   127→    <div className="flex flex-col items-center">
   128→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
   129→        {/* Background segments */}
   130→        {segments.map((seg) => (
   131→          <path
   132→            key={`${seg.from}-${seg.to}`}
   133→            d={getArcPath(seg.from, seg.to)}
   134→            fill="none"
   135→            stroke={seg.color}
   136→            strokeWidth={strokeWidth}
   137→            strokeLinecap="butt"
   138→            opacity={0.2}
   139→          />
   140→        ))}
   141→
   142→        {/* Active arc up to value */}
   143→        <motion.path
   144→          d={getArcPath(0, clamped)}
   145→          fill="none"
   146→          stroke={zone.color}
   147→          strokeWidth={strokeWidth}
   148→          strokeLinecap="round"
   149→          initial={{ pathLength: 0 }}
   150→          animate={{ pathLength: 1 }}
   151→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   152→          style={{ filter: `drop-shadow(0 0 8px ${zone.color}60)` }}
   153→        />
   154→
   155→        {/* Needle */}
   156→        <motion.line
   157→          x1={cx}
   158→          y1={cy}
   159→          x2={needleTipX}
   160→          y2={needleTipY}
   161→          stroke={zone.color}
   162→          strokeWidth={2.5}
   163→          strokeLinecap="round"
   164→          initial={{ opacity: 0 }}
   165→          animate={{ opacity: 1, x2: needleTipX, y2: needleTipY }}
   166→          transition={{ duration: 1, delay: 0.5 }}
   167→          style={{ filter: `drop-shadow(0 0 4px ${zone.color}80)` }}
   168→        />
   169→        {/* Center dot */}
   170→        <circle cx={cx} cy={cy} r={5} fill={zone.color} />
   171→
   172→        {/* Scale labels */}
   173→        <text x={startX - 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">0</text>
   174→        <text x={endX + 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">100</text>
   175→      </svg>
   176→      <div className="text-center -mt-1">
   177→        <div className="text-3xl font-bold" style={{ color: zone.color }}>
   178→          {Math.round(clamped)}
   179→        </div>
   180→        <div className="text-xs font-medium mt-0.5" style={{ color: zone.color }}>
   181→          {zone.label}
   182→        </div>
   183→        <div className="text-[10px] text-slate-500 mt-1 uppercase tracking-widest">Fear &amp; Greed Index</div>
   184→      </div>
   185→    </div>
   186→  );
   187→}
