     8→// ── In-memory Cache ──────────────────────────────────────────────
     9→interface CacheEntry<T> {
    10→  data: T;
    11→  timestamp: number;
    12→}
    13→
    14→const cache = new Map<string, CacheEntry<unknown>>();
    15→
    16→function getCached<T>(key: string, ttlMs: number): T | null {
    17→  const entry = cache.get(key);
    18→  if (entry && Date.now() - entry.timestamp < ttlMs) return entry.data as T;
    19→  cache.delete(key);
    20→  return null;
    21→}
    22→
    23→function setCache<T>(key: string, data: T): void {
    24→  if (cache.size > 300) {
    25→    const oldest = [...cache.entries()].sort((a, b) => a[1].timestamp - b[1].timestamp);
    26→    for (let i = 0; i < 80; i++) cache.delete(oldest[i][0]);
    27→  }
    28→  cache.set(key, { data, timestamp: Date.now() });
    29→}

    39→// ── Web Search ──────────────────────────────────────────────────
    40→export interface SearchResult {
    41→  url: string;
    42→  name: string;
    43→  snippet: string;
    44→  host_name: string;
    45→  rank: number;
    46→  date: string;
    47→  favicon: string;
    48→}
    49→
    50→export async function webSearch(
    51→  query: string,
    52→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    53→): Promise<SearchResult[]> {
    54→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    55→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    56→
    57→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    58→  if (cached) return cached;
    59→
    60→  const zai = await getZAI();
    61→  const args: Record<string, unknown> = { query, num };
    62→  if (recencyDays) args.recency_days = recencyDays;
    63→
    64→  let lastError: Error | null = null;
    65→  for (let attempt = 0; attempt < 3; attempt++) {
    66→    try {
    67→      const results = await zai.functions.invoke('web_search', args);
    68→      const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    69→      setCache(cacheKey, typed);
    70→      return typed;
    71→    } catch (err) {
    72→      lastError = err as Error;
    73→      const isRateLimit = (err as Error).message?.includes('429');
    74→      if (isRateLimit && attempt < 2) {
    75→        await new Promise((resolve) => setTimeout(resolve, 2000 * (attempt + 1)));
    76→        continue;
    77→      }
    78→      throw err;
    79→    }
    80→  }
    81→  throw lastError || new Error('web_search failed after retries');
    82→}

   109→// ── LLM Chat ────────────────────────────────────────────────────
   110→export async function llmChat(
   111→  systemPrompt: string,
   112→  userMessage: string,
   113→  options?: { cacheTtlMs?: number; cacheKey?: string }
   114→): Promise<string> {
   115→  const { cacheTtlMs = 600_000, cacheKey } = options || {};
   116→
   117→  if (cacheKey) {
   118→    const cached = getCached<string>(cacheKey, cacheTtlMs);
   119→    if (cached) return cached;
   120→  }
   121→
   122→  const zai = await getZAI();
   123→  const completion = await zai.chat.completions.create({
   124→    messages: [
   125→      { role: 'assistant', content: systemPrompt },
   126→      { role: 'user', content: userMessage },
   127→    ],
   128→    thinking: { type: 'disabled' },
   129→  });
   130→
   131→  const response = completion.choices[0]?.message?.content || '';
   132→  if (cacheKey) setCache(cacheKey, response);
   133→  return response;
   134→}

   150→// ── Helpers ─────────────────────────────────────────────────────
   151→function toNum(val: unknown, fallback = 0): number {
   152→  if (typeof val === 'number') return val;
   153→  if (typeof val === 'string') {
   154→    const parsed = parseFloat(val.replace(/[^0-9.\-]/g, ''));
   155→    return isNaN(parsed) ? fallback : parsed;
   156→  }
   157→  return fallback;
   158→}

   160→function clamp(val: number, min: number, max: number): number {
   161→  return Math.min(max, Math.max(min, val));

   164→function parseJSON<T>(text: string): T | null {
   165→  try {
   166→    const jsonStr = text.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
   167→    return JSON.parse(jsonStr) as T;
   168→  } catch {
   169→    return null;
   170→  }

   483→      },
   484→    };

   491→  }

   749→  symbol: string,
   750→  currentPrice: number,
   751→  timeframe: string,
   752→  indicators: Record<string, number | undefined>
   753→): Promise<{
   754→  points: Array<{ date: string; predicted: number; lower: number; upper: number; confidence: number }>;
   755→  summary: {

   757→    expectedChange: number;
   758→    volatility: number;
   759→    confidence: number;

   762→} | null> {
   763→  const cacheKey = `forecast:${symbol}:${timeframe}:${currentPrice.toFixed(2)}`;
   764→  const cached = getCached<ReturnType<typeof fetchRealForecast>>(cacheKey, 900_000);
   765→  if (cached) return cached;
   766→
   767→  try {
   768→    const companyName = getCompanyName(symbol);
   769→    const days = timeframe === 'tomorrow' ? 1 : timeframe === '3days' ? 3 : timeframe === '7days' ? 7 : 30;
   770→
   771→    const systemPrompt = `You are an AI stock forecaster. Generate a realistic price forecast for ${symbol} (${companyName}).
   772→
   773→Current data:
   774→- Price: $${currentPrice.toFixed(2)}
   775→- RSI: ${indicators.rsi?.toFixed(1) || 'N/A'}

   779→- ATR: ${indicators.atr?.toFixed(2) || 'N/A'}
   780→
   781→Timeframe: Next ${days} trading day(s).
   782→
   783→Return ONLY valid JSON:
   784→{
   785→  "points": [
   786→    {"date": "YYYY-MM-DD", "predicted": <number>, "lower": <number>, "upper": <number>, "confidence": <number 40-90>}
   787→  ],
   788→  "summary": {
   789→    "predictedPrice": <number>,
   790→    "expectedChange": <percent number>,
   791→    "volatility": <number>,
   792→    "confidence": <number>,
   793→    "analysis": "<2-3 sentence analysis>"
   794→  }
   795→}
   796→
   797→Rules:
   798→- Generate exactly ${days} forecast point(s)
   799→- Predicted price must be within 5% of current price for short timeframes
   800→- Confidence intervals should widen with time
   801→- Use realistic volatility based on ATR
   802→- Dates must be future trading days (skip weekends)
   803→Respond with JSON only.`;
   804→
   805→    const llmResponse = await llmChat(systemPrompt, `Generate ${days}-day forecast for ${symbol}.`);
   806→    const parsed = parseJSON<Record<string, unknown>>(llmResponse);
   807→    if (!parsed) return null;
   808→
   809→    const points = Array.isArray(parsed.points)
   810→      ? (parsed.points as Array<Record<string, unknown>>).map((p) => ({
   811→          date: String(p.date || ''),
   812→          predicted: clamp(toNum(p.predicted, currentPrice), currentPrice * 0.9, currentPrice * 1.1),
   813→          lower: clamp(toNum(p.lower, currentPrice), currentPrice * 0.9, currentPrice),

   815→          confidence: clamp(toNum(p.confidence, 60), 30, 90),
   816→        }))
   817→      : [];
   818→
   819→    const summary = parsed.summary as Record<string, unknown> | undefined;
   820→    const predictedPrice = points.length > 0 ? points[points.length - 1].predicted : currentPrice;
   821→    const expectedChange = ((predictedPrice - currentPrice) / currentPrice) * 100;
   822→
   823→    const result = {
   824→      points,
   825→      summary: {
   826→        predictedPrice,
   827→        expectedChange: parseFloat(expectedChange.toFixed(2)),
   828→        volatility: toNum(summary?.volatility, 2),
   829→        confidence: clamp(toNum(summary?.confidence, 60), 30, 90),
   830→        analysis: String(summary?.analysis || 'No analysis available.'),
   831→      },
   832→    };

   834→    setCache(cacheKey, result);
   835→    return result;
   836→  } catch (error) {
   837→    console.error(`Failed to fetch forecast for ${symbol}:`, error);
   838→    return null;
   839→  }
