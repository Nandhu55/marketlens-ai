     8→     8→// ── In-memory Cache ──────────────────────────────────────────────
     9→     9→interface CacheEntry<T> {
    10→    10→  data: T;
    11→    11→  timestamp: number;
    12→    12→}
    13→    13→
    14→    14→const cache = new Map<string, CacheEntry<unknown>>();
    15→    15→
    16→    16→function getCached<T>(key: string, ttlMs: number): T | null {
    17→    17→  const entry = cache.get(key);
    18→    18→  if (entry && Date.now() - entry.timestamp < ttlMs) return entry.data as T;
    19→    19→  cache.delete(key);
    20→    20→  return null;
    21→    21→}
    22→    22→
    23→    23→function setCache<T>(key: string, data: T): void {
    24→    24→  if (cache.size > 300) {
    25→    25→    const oldest = [...cache.entries()].sort((a, b) => a[1].timestamp - b[1].timestamp);
    26→    26→    for (let i = 0; i < 80; i++) cache.delete(oldest[i][0]);
    27→    27→  }
    28→    28→  cache.set(key, { data, timestamp: Date.now() });
    29→    29→}

    40→    40→export interface SearchResult {
    41→    41→  url: string;
    42→    42→  name: string;
    43→    43→  snippet: string;
    44→    44→  host_name: string;

    49→    49→
    50→    50→export async function webSearch(
    51→    51→  query: string,
    52→    52→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    53→    53→): Promise<SearchResult[]> {
    54→    54→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    55→    55→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    56→    56→
    57→    57→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    58→    58→  if (cached) return cached;
    59→    59→
    60→    60→  const zai = await getZAI();
    61→    61→  const args: Record<string, unknown> = { query, num };
    62→    62→  if (recencyDays) args.recency_days = recencyDays;
    63→    63→
    64→    64→  let lastError: Error | null = null;
    65→    65→  for (let attempt = 0; attempt < 3; attempt++) {
    66→    66→    try {
    67→    67→      const results = await zai.functions.invoke('web_search', args);
    68→    68→      const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    69→    69→      setCache(cacheKey, typed);
    70→    70→      return typed;
    71→    71→    } catch (err) {
    72→    72→      lastError = err as Error;
    73→    73→      const isRateLimit = (err as Error).message?.includes('429');
    74→    74→      if (isRateLimit && attempt < 2) {
    75→    75→        await new Promise((resolve) => setTimeout(resolve, 2000 * (attempt + 1)));
    76→    76→        continue;
    77→    77→      }
    78→    78→      throw err;
    79→    79→    }
    80→    80→  }
    81→    81→  throw lastError || new Error('web_search failed after retries');
    82→    82→}
