   110→// ─── Custom Candlestick Shape ────────────────────────────────────────
   111→function CandlestickBar(props: any) {
   112→  const { x, y, width, height, payload } = props;
   113→  if (!payload || isNaN(payload.body[0]) || isNaN(payload.body[1])) return null;
   114→
   115→  const open = payload.open;
   116→  const close = payload.close;
   117→  const low = payload.low;
   118→  const high = payload.high;
   119→  const isBull = close >= open;
   120→  const color = isBull ? CHART_COLORS.bull : CHART_COLORS.bear;
   121→
   122→  const barWidth = Math.max(width * 0.6, 3);
   123→  const cx = x + width / 2;
   124→  const bodyTop = y;
   125→  const bodyBottom = y + height;
   126→  const bodyHeight = Math.max(height, 1);
   127→
   128→  // Calculate y-axis scale from the chart dimensions
   129→  const priceRange = high - low;
   130→  const totalHeight = props.backgroundHeight || 300;
   131→  const yScale = totalHeight / ((props.yAxisMax || high) - (props.yAxisMin || low));
   132→
   133→  const highY = bodyTop - (high - Math.max(open, close)) * yScale;
   134→  const lowY = bodyBottom + (Math.min(open, close) - low) * yScale;
   135→
   136→  return (
   137→    <g>
   138→      {/* Wick */}
   139→      <line
   140→        x1={cx}
   141→        y1={lowY}
   142→        x2={cx}
   143→        y2={highY}
   144→        stroke={color}
   145→        strokeWidth={1}
   146→        opacity={0.7}
   147→      />
   148→      {/* Body */}
   149→      <rect
   150→        x={cx - barWidth / 2}
   151→        y={bodyTop}
   152→        width={barWidth}
   153→        height={bodyHeight}
   154→        fill={isBull ? color : color}
   155→        opacity={isBull ? 0.9 : 0.9}
   156→        stroke={color}
   157→        strokeWidth={0.5}
   158→        rx={1}
   159→      />
   160→    </g>
   161→  );
   162→}

   282→export function LiveMarketPage() {
   283→  const { selectedStock, setSelectedStock, searchHistory, addSearchHistory } = useAppStore();
   284→  const [searchQuery, setSearchQuery] = useState('');
   285→  const [showDropdown, setShowDropdown] = useState(false);
   286→  const [isLoading, setIsLoading] = useState(true);
   287→  const [activeTab, setActiveTab] = useState('ma');
   288→  const searchRef = useRef<HTMLDivElement>(null);
   289→  const dropdownRef = useRef<HTMLDivElement>(null);
   290→
   291→  // ─── Simulated loading ─────────────────────────────────────────────
   292→  useEffect(() => {
   293→    const timer = setTimeout(() => setIsLoading(false), 300);
   294→    return () => {
   295→      clearTimeout(timer);
   296→      setIsLoading(true);
   297→    };
   298→  }, [selectedStock.symbol]);
   299→
   300→  // ─── Click outside to close dropdown ───────────────────────────────
   301→  useEffect(() => {
   302→    const handleClickOutside = (e: MouseEvent) => {
   303→      if (
   304→        searchRef.current &&
   305→        !searchRef.current.contains(e.target as Node) &&
   306→        dropdownRef.current &&
   307→        !dropdownRef.current.contains(e.target as Node)
   308→      ) {
   309→        setShowDropdown(false);
   310→      }
   311→    };
   312→    document.addEventListener('mousedown', handleClickOutside);
   313→    return () => document.removeEventListener('mousedown', handleClickOutside);
   314→  }, []);
   315→
   316→  // ─── Generate stock data ───────────────────────────────────────────
   317→  const stockData = useMemo(() => generateStockData(selectedStock.symbol, 90), [selectedStock.symbol]);
   318→  const closes = useMemo(() => stockData.map((d) => d.close), [stockData]);
   319→

   321→  const candleData = useMemo(
   322→    () =>
   323→      stockData.map((d) => ({
   324→        ...d,
   325→        bullish: d.close >= d.open,
   326→        body: [Math.min(d.open, d.close), Math.max(d.open, d.close)],
   327→        wick: [d.low, d.high],
   328→      })),
   329→    [stockData]
   330→  );
   331→
   332→  // ─── Volume data ───────────────────────────────────────────────────
   333→  const volumeData = useMemo(
   334→    () =>
   335→      stockData.map((d) => ({
   336→        date: d.date,
   337→        volume: d.volume,
   338→        bullish: d.close >= d.open,
   339→      })),
   340→    [stockData]
   341→  );
   342→
   343→  // ─── Moving Averages data ──────────────────────────────────────────
   344→  const maData = useMemo(() => {
   345→    const sma20 = calculateSMA(closes, 20);
   346→    const ema20 = calculateEMA(closes, 20);
   347→    const sma50 = calculateSMA(closes, 50);
   348→    return stockData.map((d, i) => ({
   349→      date: d.date,
   350→      price: d.close,
   351→      sma20: sma20[i],
   352→      ema20: ema20[i],
   353→      sma50: sma50[i],
   354→    }));
   355→  }, [stockData, closes]);
   356→
   357→  // ─── RSI data ──────────────────────────────────────────────────────
   358→  const rsiData = useMemo(() => {
   359→    const rsi = calculateRSI(closes, 14);
   360→    return stockData.map((d, i) => ({
   361→      date: d.date,
   362→      rsi: rsi[i],
   363→    }));
   364→  }, [stockData, closes]);
   365→
   366→  // ─── MACD data ─────────────────────────────────────────────────────
   367→  const macdData = useMemo(() => {
   368→    const { macd, signal, histogram } = calculateMACD(closes, 12, 26, 9);
   369→    return stockData.map((d, i) => ({
   370→      date: d.date,
   371→      macd: macd[i],
   372→      signal: signal[i - (macd.length - signal.length)] ?? NaN,
   373→      histogram: histogram[i],
   374→    }));
   375→  }, [stockData, closes]);
   376→
   377→  // ─── Bollinger Bands data ──────────────────────────────────────────
   378→  const bbData = useMemo(() => {
   379→    const { upper, middle, lower } = calculateBollingerBands(closes, 20, 2);
   380→    return stockData.map((d, i) => ({
   381→      date: d.date,
   382→      price: d.close,
   383→      upper: upper[i],
   384→      middle: middle[i],
   385→      lower: lower[i],
   386→    }));
   387→  }, [stockData, closes]);
   388→
   389→  // ─── Sidebar sparkline data (last 30 days) ─────────────────────────
   390→  const sparkData = useMemo(() => {
   391→    const last30 = stockData.slice(-30);
   392→    return last30.map((d) => ({ date: d.date, close: d.close }));
   393→  }, [stockData]);

   395→  // ─── Stock info ────────────────────────────────────────────────────
   396→  const stockInfo = useMemo(() => STOCKS[selectedStock.symbol], [selectedStock.symbol]);

   415→        setSelectedStock({
   416→          symbol: stock.symbol,
   417→          name: stock.name,
   418→          price: stock.price,
   419→          change: stock.change,
   420→          changePercent: stock.changePercent,
   421→          volume: stock.volume,
   422→          marketCap: stock.marketCap,
   423→        });
   424→        addSearchHistory(symbol);
   425→        setSearchQuery('');
   426→        setShowDropdown(false);
   427→      }
   428→    },
   429→    [setSelectedStock, addSearchHistory]
   430→  );
   431→
   432→  // ─── Handle recent search click ────────────────────────────────────

   452→    () => Math.max(...stockData.map((d) => d.volume)) * 1.1,

   532→                      >
   533→                        <div className="flex-1 min-w-0">
   534→                          <div className="flex items-center gap-2">
   535→                            <span className="font-bold text-white text-sm">{stock.symbol}</span>

   538→                          <span className="text-[10px] text-slate-600">{stock.sector}</span>
