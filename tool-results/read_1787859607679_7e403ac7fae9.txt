   104→// ─── Chart Tooltip ───────────────────────────────────────────────────
   105→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string }>; label?: string }) {
   106→  if (!active || !payload?.length) return null;
   107→  return (
   108→    <div className="market-tooltip px-3 py-2 text-xs">
   109→      <p className="text-slate-400 mb-1">{label}</p>
   110→      <p className="text-white font-semibold">${payload[0].value.toFixed(2)}</p>
   111→    </div>
   112→  );
   113→}

   115→// ─── Fear & Greed Semicircle Gauge ───────────────────────────────────
   116→function FearGreedGauge({ value }: { value: number }) {
   117→  if (value <= 0) {
   118→    return (
   119→      <div className="flex flex-col items-center justify-center h-full">
   120→        <p className="text-slate-500 text-sm">No Data</p>
   121→      </div>
   122→    );
   123→  }
   124→
   125→  const clamped = Math.max(0, Math.min(100, value));
   126→
   127→  const getZone = () => {
   128→    if (clamped <= 25) return { label: 'Extreme Fear', color: '#ef4444' };
   129→    if (clamped <= 45) return { label: 'Fear', color: '#f97316' };
   130→    if (clamped <= 55) return { label: 'Neutral', color: '#eab308' };
   131→    if (clamped <= 75) return { label: 'Greed', color: '#22c55e' };
   132→    return { label: 'Extreme Greed', color: '#10b981' };
   133→  };
   134→
   135→  const zone = getZone();
   136→  const width = 260;
   137→  const height = 145;
   138→  const cx = width / 2;
   139→  const cy = height - 10;
   140→  const radius = 100;
   141→  const strokeWidth = 18;
   142→  const valueAngle = Math.PI - (clamped / 100) * Math.PI;
   143→  const needleLen = radius - strokeWidth - 5;
   144→  const needleTipX = cx - needleLen * Math.cos(valueAngle);
   145→  const needleTipY = cy - needleLen * Math.sin(valueAngle);
   146→  const startX = cx - radius;
   147→  const endX = cx + radius;
   148→
   149→  const segments = [
   150→    { from: 0, to: 25, color: '#ef4444' },
   151→    { from: 25, to: 45, color: '#f97316' },
   152→    { from: 45, to: 55, color: '#eab308' },
   153→    { from: 55, to: 75, color: '#22c55e' },
   154→    { from: 75, to: 100, color: '#10b981' },
   155→  ];
   156→
   157→  const getArcPath = (fromPct: number, toPct: number) => {
   158→    const startAngle = Math.PI - (fromPct / 100) * Math.PI;
   159→    const endAngle = Math.PI - (toPct / 100) * Math.PI;
   160→    const sx = cx + radius * Math.cos(startAngle);
   161→    const sy = cy - radius * Math.sin(startAngle);
   162→    const ex = cx + radius * Math.cos(endAngle);
   163→    const ey = cy - radius * Math.sin(endAngle);
   164→    const largeArc = (fromPct - toPct) > 50 ? 1 : 0;
   165→    return `M ${sx} ${sy} A ${radius} ${radius} 0 ${largeArc} 0 ${ex} ${ey}`;
   166→  };
   167→
   168→  return (
   169→    <div className="flex flex-col items-center">
   170→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
   171→        {segments.map((seg) => (
   172→          <path
   173→            key={`${seg.from}-${seg.to}`}
   174→            d={getArcPath(seg.from, seg.to)}
   175→            fill="none"
   176→            stroke={seg.color}
   177→            strokeWidth={strokeWidth}
   178→            strokeLinecap="butt"
   179→            opacity={0.2}
   180→          />
   181→        ))}
   182→        <motion.path
   183→          d={getArcPath(0, clamped)}
   184→          fill="none"
   185→          stroke={zone.color}
   186→          strokeWidth={strokeWidth}
   187→          strokeLinecap="round"
   188→          initial={{ pathLength: 0 }}
   189→          animate={{ pathLength: 1 }}
   190→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   191→          style={{ filter: `drop-shadow(0 0 8px ${zone.color}60)` }}
   192→        />
   193→        <motion.line
   194→          x1={cx} y1={cy} x2={needleTipX} y2={needleTipY}
   195→          stroke={zone.color} strokeWidth={2.5} strokeLinecap="round"
   196→          initial={{ opacity: 0 }}
   197→          animate={{ opacity: 1, x2: needleTipX, y2: needleTipY }}
   198→          transition={{ duration: 1, delay: 0.5 }}
   199→          style={{ filter: `drop-shadow(0 0 4px ${zone.color}80)` }}
   200→        />
   201→        <circle cx={cx} cy={cy} r={5} fill={zone.color} />
   202→        <text x={startX - 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">0</text>
   203→        <text x={endX + 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">100</text>
   204→      </svg>
   205→      <div className="text-center -mt-1">
   206→        <div className="text-3xl font-bold" style={{ color: zone.color }}>
   207→          {Math.round(clamped)}
   208→        </div>
   209→        <div className="text-xs font-medium mt-0.5" style={{ color: zone.color }}>
   210→          {zone.label}
   211→        </div>
   212→        <div className="text-[10px] text-slate-500 mt-1 uppercase tracking-widest">Fear &amp; Greed Index</div>
   213→      </div>
   214→    </div>
   215→  );
   216→}

   218→// ─── Ticker item component ───────────────────────────────────────────
   219→function TickerItem({ symbol, price, changePercent, isLive }: { symbol: string; price: number; changePercent: number; isLive?: boolean }) {
   220→  const isPositive = changePercent >= 0;
   221→  return (
   222→    <div className="flex items-center gap-2.5 px-5 py-2 whitespace-nowrap">
   223→      {isLive && <span className="inline-block w-1.5 h-1.5 rounded-full bg-emerald-500" />}
   224→      <span className="text-xs font-bold text-white tracking-wider">{symbol}</span>
   225→      <span className="text-xs text-slate-300">{isIndianSymbol(symbol) ? `₹${formatPrice(price)}` : `$${formatPrice(price)}`}</span>
   226→      <span
   227→        className={cn(
   228→          'text-xs font-semibold flex items-center gap-0.5',
   229→          isPositive ? 'text-emerald-400' : 'text-red-400'
   230→        )}
   231→      >
   232→        {isPositive ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
   233→        {formatPercent(changePercent)}
   234→      </span>
   235→    </div>
   236→  );
   237→}

   253→// ─── Main Dashboard Page ────────────────────────────────────────────
   254→export function DashboardPage() {
   255→  const { realQuotes, selectedStock, watchlist, isLive, liveTicks, wsConnectionStatus, staleSymbols } = useAppStore();
   256→  const [overviewData, setOverviewData] = useState<MarketOverviewResponse | null>(null);
   257→  const [stockData, setStockData] = useState<StockDataResponse | null>(null);
   258→  const [loading, setLoading] = useState(true);
   259→  const [stockLoading, setStockLoading] = useState(true);
   260→  const [error, setError] = useState(false);
   261→
   262→  const fetchMarketOverview = useCallback(async () => {
   263→    setLoading(true);
   264→    setError(false);
   265→    try {
   266→      const res = await fetch('/api/market-overview');
   267→      const data: MarketOverviewResponse = await res.json();
   268→      setOverviewData(data);
   269→    } catch {
   270→      setError(true);
   271→    } finally {
   272→      setLoading(false);
   273→    }
   274→  }, []);
   275→
   276→  const fetchStockData = useCallback(async () => {
   277→    setStockLoading(true);
   278→    try {
   279→      const res = await fetch(`/api/stock?symbol=${selectedStock.symbol}&days=30`);
   280→      const data: StockDataResponse = await res.json();
   281→      setStockData(data);
   282→    } catch {
   283→      setStockData(null);
   284→    } finally {
   285→      setStockLoading(false);
   286→    }
   287→  }, [selectedStock.symbol]);
   288→
   289→  useEffect(() => {
   290→    fetchMarketOverview();
   291→  }, [fetchMarketOverview]);
   292→
   293→  useEffect(() => {
   294→    fetchStockData();
   295→  }, [fetchStockData]);
   296→
   297→  // Merge REST quotes with live WebSocket ticks
   298→  // Live ticks take priority for subscribed symbols
   299→  const mergedQuotes: Record<string, { price: number; changePercent: number; source: string; isLive: boolean }> = {};
   300→  // Start with REST quotes
   301→  for (const [symbol, q] of Object.entries(realQuotes)) {
   302→    mergedQuotes[symbol] = { price: q.price, changePercent: q.changePercent, source: q.source, isLive: false };
   303→  }
   304→  // Overlay with live ticks
   305→  for (const [symbol, tick] of Object.entries(liveTicks)) {
   306→    if (tick.dataStatus === 'LIVE' && !staleSymbols.includes(symbol)) {
   307→      mergedQuotes[symbol] = {
   308→        price: tick.price,
   309→        changePercent: tick.changePercent,
   310→        source: 'upstox-ws',
   311→        isLive: true,
   312→      };
   313→    }
   314→  }
   315→
   316→  const tickerItems = Object.entries(mergedQuotes).map(([symbol, q]) => ({
   317→    symbol,
   318→    price: q.price,
   319→    changePercent: q.changePercent,
   320→    isLive: q.isLive,
   321→  }));
   322→
   323→  // Chart data from stock OHLCV
   324→  const chartData = stockData?.ohlcv?.map((d) => ({
   325→    date: d.date.slice(5),
   326→    price: d.close,
   327→  })) || [];
   328→
   329→  // Aggregated values from overview
   330→  const totalVolume = overviewData?.quotes
   331→    ? Object.values(overviewData.quotes).reduce((sum, q) => sum + (q.volume || 0), 0)
   332→    : 0;
   333→
   334→  const avgChange = overviewData?.quotes
   335→    ? (() => {
   336→        const vals = Object.values(overviewData.quotes).map((q) => q.changePercent);
   337→        return vals.length > 0 ? vals.reduce((a, b) => a + b, 0) / vals.length : 0;
   338→      })()
   339→    : 0;
   340→
   341→  const topGainer = overviewData?.overview?.topGainers?.[0];
   342→  const topLoser = overviewData?.overview?.topLosers?.[0];
   343→
   344→  const hasRealQuotes = tickerItems.length > 0;
   345→  const hasOverview = !!overviewData?.hasRealData;
   346→  const fearGreedValue = overviewData?.overview?.fearGreedIndex || 0;
   347→
   348→  return (
   349→    <motion.div
   350→      variants={containerVariants}
   351→      initial="hidden"
   352→      animate="show"
   353→      className="space-y-6"
   354→    >
   355→      {/* ── Market Ticker Bar ─────────────────────────────────── */}
   356→      {hasRealQuotes ? (
   357→        <motion.div variants={itemVariants} className="glass-card overflow-hidden">
   358→          <div className="flex items-center">
   359→            <div className="flex-shrink-0 px-4 py-2 border-r border-white/5 flex items-center gap-2">
   360→              {(isLive || wsConnectionStatus === 'CONNECTED') && (
   361→                <span className="flex items-center gap-1.5 text-[10px] font-semibold text-emerald-400 uppercase tracking-wider">
   362→                  <Zap className="w-3 h-3" />
   363→                  {wsConnectionStatus === 'CONNECTED' ? 'Live' : 'Loading'}
   364→                </span>
   365→              )}
   366→              <CandlestickChart className="w-3.5 h-3.5 text-indigo-400" />
   367→            </div>
   368→            <div className="overflow-hidden flex-1">
   369→              <div className="ticker-scroll flex">
   370→                {[...tickerItems, ...tickerItems].map((item, i) => (
   371→                  <TickerItem key={`${item.symbol}-${i}`} {...item} />
   372→                ))}
   373→              </div>
   374→            </div>
   375→          </div>
   376→        </motion.div>
   377→      ) : (
   378→        <motion.div variants={itemVariants} className="glass-card overflow-hidden">
   379→          <div className="flex items-center justify-center px-4 py-3">
   380→            <p className="text-sm text-slate-500">Waiting for live market data...</p>
   381→          </div>
   382→        </motion.div>
   383→      )}
   384→
   385→      {/* ── KPI Cards ──────────────────────────────────────────── */}
   386→      <motion.div variants={itemVariants}>
   387→        {loading ? (
   388→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   389→            {Array.from({ length: 8 }).map((_, i) => (
   390→              <KPISkeleton key={i} />
   391→            ))}
   392→          </div>
   393→        ) : error ? (
   394→          <div className="flex flex-col items-center justify-center py-12">
   395→            <EmptyState
   396→              icon={<Activity className="w-10 h-10" />}
   397→              title="No Live Data Available"
   398→              description="Could not fetch market overview. Please try again."
   399→              action={
   400→                <Button variant="outline" size="sm" onClick={fetchMarketOverview} className="gradient-button border-0">
   401→                  <RefreshCw className="w-3.5 h-3.5 mr-2" />
   402→                  Refresh
   403→                </Button>
   404→              }
   405→            />
   406→          </div>
   407→        ) : (
   408→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   409→            <KPICard
   410→              title="Total Volume"
   411→              value={totalVolume > 0 ? formatNumber(totalVolume) : 'No Data'}
   412→              icon={BarChart3}
   413→              iconColor="text-cyan-400"
   414→              glowColor="glow-cyan"
   415→              index={0}
   416→            />
   417→            <KPICard

   420→              change={overviewData && totalVolume > 0 ? formatPercent(avgChange) : undefined}
   421→              changeType={avgChange >= 0 ? 'positive' : 'negative'}
   422→              icon={TrendingUp}
   423→              iconColor="text-emerald-400"
   424→              glowColor="glow-green"

   426→            />
   427→            <KPICard

   430→              change={topGainer ? `+${topGainer.change.toFixed(2)}%` : undefined}
   431→              changeType="positive"
   432→              icon={ArrowUpRight}
   433→              iconColor="text-emerald-400"
   434→              glowColor="glow-green"
   435→              index={2}
   436→            />
   437→            <KPICard
