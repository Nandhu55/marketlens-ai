    59→    59→// ─── Stochastic Oscillator ────────────────────────────────────────────────
    60→    60→function calculateStochastic(data: OHLCV[], kPeriod: number = 14, dPeriod: number = 3): { k: number[]; d: number[] } {
    61→    61→  const kValues: number[] = [];
    62→    62→
    63→    63→  for (let i = 0; i < data.length; i++) {
    64→    64→    if (i < kPeriod - 1) {
    65→    65→      kValues.push(NaN);
    66→    66→      continue;
    67→    67→    }
    68→    68→    const slice = data.slice(i - kPeriod + 1, i + 1);
    69→    69→    const highest = Math.max(...slice.map((d) => d.high));
    70→    70→    const lowest = Math.min(...slice.map((d) => d.low));
    71→    71→    const range = highest - lowest;
    72→    72→    const k = range === 0 ? 50 : ((data[i].close - lowest) / range) * 100;
    73→    73→    kValues.push(parseFloat(k.toFixed(2)));
    74→    74→  }
    75→    75→
    76→    76→  const dValues = calculateSMA(kValues.filter((v) => !isNaN(v)), dPeriod);
    77→    77→  const dPadded: number[] = new Array(kValues.length - dValues.length).fill(NaN).concat(dValues);
    78→    78→
    79→    79→  return { k: kValues, d: dPadded };
    80→    80→}

   106→   106→function DarkTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color: string }>; label?: string }) {
   107→   107→  if (!active || !payload?.length) return null;
   108→   108→  return (
   109→   109→    <div className="glass p-3 rounded-lg border border-indigo-500/20 min-w-[140px]">
   110→   110→      <p className="text-xs text-slate-400 mb-2">{label}</p>
   111→   111→      {payload.map((entry) => (
   112→   112→        <div key={entry.dataKey} className="flex items-center gap-2 text-sm mb-0.5 last:mb-0">
   113→   113→          <div className="w-2.5 h-2.5 rounded-full flex-shrink-0" style={{ backgroundColor: entry.color }} />
   114→   114→          <span className="text-slate-400 capitalize text-xs">{entry.dataKey}:</span>
   115→   115→          <span className="font-semibold text-white text-xs">{typeof entry.value === 'number' ? entry.value.toFixed(2) : entry.value}</span>
   116→   116→        </div>
   117→   117→      ))}
   118→   118→    </div>
   119→   119→  );
   120→   120→}
