    45→// ─── Seeded PRNG based on symbol ──────────────────────────────────────────
    46→function seededRandom(seed: number): () => number {
    47→  let s = seed;
    48→  return () => {
    49→    s = (s * 16807 + 0) % 2147483647;
    50→    return (s - 1) / 2147483646;
    51→  };
    52→}
    53→
    54→function symbolSeed(symbol: string): number {
    55→  return symbol.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0) + 42;
    56→}

    58→// ─── Generate prediction data deterministically ───────────────────────────
    59→interface PredictionData {
    60→  bullishProb: number;
    61→  bearishProb: number;
    62→  volatilityScore: number;
    63→  confidenceScore: number;
    64→  upperBound: number;
    65→  lowerBound: number;
    66→  recommendation: 'BUY' | 'HOLD' | 'SELL';
    67→  riskLevel: 'Low' | 'Medium' | 'High' | 'Extreme';
    68→  accuracy: number;
    69→  precision: number;
    70→  recall: number;
    71→  f1Score: number;
    72→  rmse: number;
    73→  mae: number;
    74→}
    75→
    76→function generatePrediction(symbol: string, currentPrice: number): PredictionData {
    77→  const rng = seededRandom(symbolSeed(symbol));
    78→
    79→  const bullishBase = 35 + rng() * 40;
    80→  const bullishProb = parseFloat(bullishBase.toFixed(1));
    81→  const bearishProb = parseFloat((100 - bullishProb).toFixed(1));
    82→
    83→  const volatilityScore = parseFloat((20 + rng() * 60).toFixed(1));
    84→  const confidenceScore = parseFloat((65 + rng() * 30).toFixed(1));
    85→
    86→  const spread = currentPrice * (0.02 + rng() * 0.06);
    87→  const upperBound = parseFloat((currentPrice + spread * (0.4 + rng() * 0.6)).toFixed(2));
    88→  const lowerBound = parseFloat((currentPrice - spread * (0.4 + rng() * 0.6)).toFixed(2));
    89→
    90→  let recommendation: 'BUY' | 'HOLD' | 'SELL';
    91→  if (bullishProb > 62) recommendation = 'BUY';
    92→  else if (bullishProb < 38) recommendation = 'SELL';
    93→  else recommendation = 'HOLD';
    94→
    95→  let riskLevel: 'Low' | 'Medium' | 'High' | 'Extreme';
    96→  if (volatilityScore < 35) riskLevel = 'Low';
    97→  else if (volatilityScore < 55) riskLevel = 'Medium';
    98→  else if (volatilityScore < 75) riskLevel = 'High';
    99→  else riskLevel = 'Extreme';
   100→
   101→  return {
   102→    bullishProb,
   103→    bearishProb,
   104→    volatilityScore,
   105→    confidenceScore,
   106→    upperBound,
   107→    lowerBound,
   108→    recommendation,
   109→    riskLevel,
   110→    accuracy: parseFloat((82 + rng() * 12).toFixed(1)),
   111→    precision: parseFloat((80 + rng() * 12).toFixed(1)),
   112→    recall: parseFloat((83 + rng() * 12).toFixed(1)),
   113→    f1Score: parseFloat((81 + rng() * 12).toFixed(1)),
   114→    rmse: parseFloat((1.5 + rng() * 2.5).toFixed(2)),
   115→    mae: parseFloat((1.0 + rng() * 2.0).toFixed(2)),
   116→  };
   117→}

   119→// ─── Historical accuracy chart data ──────────────────────────────────────
   120→interface AccuracyPoint {
   121→  date: string;
   122→  predicted: number;
   123→  actual: number;
   124→}
   125→
   126→function generateAccuracyData(symbol: string, currentPrice: number): AccuracyPoint[] {
   127→  const rng = seededRandom(symbolSeed(symbol) + 999);
   128→  const data: AccuracyPoint[] = [];
   129→  let price = currentPrice * 0.92;
   130→  const now = new Date();
   131→
   132→  for (let i = 30; i >= 0; i--) {
   133→    const date = new Date(now);
   134→    date.setDate(date.getDate() - i);
   135→    if (date.getDay() === 0 || date.getDay() === 6) continue;
   136→
   137→    const change = (rng() - 0.47) * 0.03;
   138→    const actual = parseFloat((price * (1 + change)).toFixed(2));
   139→    const noise = (rng() - 0.5) * currentPrice * 0.015;
   140→    const predicted = parseFloat((actual + noise).toFixed(2));
   141→
   142→    data.push({
   143→      date: date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   144→      predicted,
   145→      actual,
   146→    });
   147→    price = actual;
   148→  }
   149→  return data;
   150→}

Currently trading at $${price.toFixed(2)} with a P/E ratio of ${pe.toFixed(1)}, the stock faces headwinds from deteriorating technical structure and declining volume support. The stock is caught between competing forces, with neither buyers nor sellers able to establish clear control. Our AI model indicates a period of price discovery where the stock is building a base ahead of the next directional move. Our model assigns a ${prediction.confidenceScore.toFixed(0)}% confidence level to this assessment.`
   188→    );

   190→    paragraphs.push(
   191→      `Pattern recognition indicates a potential breakdown scenario, with critical support at $${prediction.lowerBound.toFixed(2)} under pressure. If this level fails to hold, the next significant support zone lies approximately 3-5% lower. Resistance at $${prediction.upperBound.toFixed(2)} remains firm, and the inability to reclaim this level reinforces the bearish thesis. The MACD histogram is ${prediction.bearishProb > 65 ? Our model maintains a ${prediction.confidenceScore.toFixed(0)}% confidence level for this bearish outlook.`
   192→    );

   195→      `Price is currently consolidating between the $${prediction.lowerBound.toFixed(2)} support and $${prediction.upperBound.toFixed(2)} resistance levels. Our model maintains ${prediction.confidenceScore.toFixed(0)}% confidence in the neutral assessment.`
   196→    );
   197→  }
   198→
   199→  // Paragraph 4: Risk assessment and outlook
   200→  paragraphs.push(
   201→    `Risk management considerations suggest that position sizing should account for the ${prediction.riskLevel.toLowerCase()} volatility environment currently exhibited by ${name}. The predicted price range of $${prediction.lowerBound.toFixed(2)} to $${prediction.upperBound.toFixed(2)} represents a potential ${((prediction.upperBound - prediction.lowerBound) / price * 100).toFixed(1)}% swing over the forecast horizon. The AI model's historical accuracy of ${prediction.accuracy.toFixed(1)}% for ${symbol} predictions provides a statistical edge, though no model guarantees future performance. Monitor volume patterns and sector rotation dynamics for early warning signs of trend changes.`
   202→  );
   203→
   204→  return paragraphs;
   205→}
