    22→// ── Web Search ──────────────────────────────────────────────────
    23→export interface SearchResult {
    24→  url: string;
    25→  name: string;
    26→  snippet: string;
    27→  host_name: string;
    28→  rank: number;
    29→  date: string;
    30→  favicon: string;
    31→}
    32→
    33→export async function webSearch(
    34→  query: string,
    35→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    36→): Promise<SearchResult[]> {
    37→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    38→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    39→
    40→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    41→  if (cached) return cached;
    42→
    43→  const zai = await getZAI();
    44→  const args: Record<string, unknown> = { query, num };
    45→  if (recencyDays) args.recency_days = recencyDays;
    46→
    47→  try {
    48→    const results = await Promise.race([
    49→      zai.functions.invoke('web_search', args),
    50→      new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    51→    ]);
    52→    const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    53→    setCache(cacheKey, typed);
    54→    return typed;
    55→  } catch (err) {
    56→    throw err;
    57→  }
    58→}

    61→export async function readPage(url: string, cacheTtlMs = 600_000): Promise<{
    62→  title: string;
    63→  html: string;
    64→  url: string;
    65→  publishedTime?: string;
    66→}> {
    67→  const cacheKey = `page:${url}`;
    68→  const cached = getCached<{ title: string; html: string; url: string; publishedTime?: string }>(cacheKey, cacheTtlMs);
    69→  if (cached) return cached;
    70→
    71→  const zai = await getZAI();
    72→  const result = await Promise.race([
    73→    zai.functions.invoke('page_reader', { url }),
    74→    new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    75→  ]);
    76→
    77→  const data = {
    78→    title: result.data?.title || '',
    79→    html: result.data?.html || '',
    80→    url: result.data?.url || url,
    81→    publishedTime: result.data?.publishedTime || result.data?.publish_time,
    82→  };
    83→
    84→  setCache(cacheKey, data);
    85→  return data;
    86→}

    88→// ── LLM Chat ────────────────────────────────────────────────────
    89→export async function llmChat(
    90→  systemPrompt: string,
    91→  userMessage: string,
    92→  options?: { cacheTtlMs?: number; cacheKey?: string }
    93→): Promise<string> {
    94→  const { cacheTtlMs = 600_000, cacheKey } = options || {};
    95→
    96→  if (cacheKey) {
    97→    const cached = getCached<string>(cacheKey, cacheTtlMs);
    98→    if (cached) return cached;
    99→  }
   100→
   101→  const zai = await getZAI();
   102→  const completion = await Promise.race([
   103→    zai.chat.completions.create({
   104→      messages: [
   105→        { role: 'assistant', content: systemPrompt },
   106→        { role: 'user', content: userMessage },
   107→      ],
   108→      thinking: { type: 'disabled' },
   109→    }),
   110→    new Promise((_, reject) => setTimeout(() => reject(new Error('LLM Timeout')), 20000)),
   111→  ]) as Awaited<ReturnType<typeof zai.chat.completions.create>>;
   112→
   113→
   114→  const response = completion.choices[0]?.message?.content || '';
   115→  if (cacheKey) setCache(cacheKey, response);
   116→  return response;
   117→}

   133→// ── Helpers ─────────────────────────────────────────────────────
   134→function toNum(val: unknown, fallback = 0): number {
   135→  if (typeof val === 'number') return val;
   136→  if (typeof val === 'string') {
   137→    const parsed = parseFloat(val.replace(/[^0-9.\-]/g, ''));
   138→    return isNaN(parsed) ? fallback : parsed;
   139→  }
   140→  return fallback;
   141→}

   143→function clamp(val: number, min: number, max: number): number {
   144→  return Math.min(max, Math.max(min, val));
   145→}

   147→function parseJSON<T>(text: string): T | null {
   148→  try {
   149→    const jsonStr = text.replace(/```json\n?/g, '').replace(/```\n?/g, '').trim();
   150→    return JSON.parse(jsonStr) as T;
   151→  } catch {
   152→    return null;
   153→  }

   326→        low: c.low,

   351→async function fetchHistoricalOHLCVUS(
   352→  symbol: string,
   353→  days: number = 90
   354→): Promise<OHLCV[] | null> {
   355→  try {
   356→    const companyName = getCompanyName(symbol);
   357→
   358→    // Strategy 1: Try to read Yahoo Finance history page
   359→    const yahooUrl = `https://finance.yahoo.com/quote/${symbol}/history/`;
   360→    let pageData: { html: string; title: string } | null = null;
   361→
   362→    try {
   363→      pageData = await readPage(yahooUrl, 300_000);
   364→    } catch {
   365→      // Yahoo may block, fall through to search-based approach
   366→    }
   367→
   368→    if (pageData?.html && pageData.html.length > 500) {
   369→      const systemPrompt = `You are a financial data parser. Extract historical stock price data from the provided HTML.
   370→Return ONLY a valid JSON array of objects with exact fields: date, open, high, low, close, volume.
   371→- date format: YYYY-MM-DD
   372→- all values are numbers
   373→- Include the most recent ${days} trading days
   374→- Sort chronologically from oldest to newest
   375→- If data is not available in the HTML, return an empty array []
   376→Respond with JSON only, no markdown.`;
   377→
   378→      const htmlSnippet = pageData.html.substring(0, 8000);
   379→      const userMessage = `Extract OHLCV data for ${symbol} from this HTML:\n${htmlSnippet}`;
   380→
   381→      const llmResponse = await llmChat(systemPrompt, userMessage, {
   382→        cacheKey: `ohlcv-yahoo:${symbol}:${days}`,
   383→        cacheTtlMs: 600_000,
   384→      });
   385→
   386→      const parsed = parseJSON<OHLCV[]>(llmResponse);
   387→      if (parsed && parsed.length >= 20) {
   388→        return parsed;
   389→      }
   390→    }
   391→
   392→    // Strategy 2: Search for historical price data pages
   393→    const searchQuery = `${symbol} ${companyName} stock price history last ${days} days OHLCV data 2025`;

   404→Most recent ${days} days. Chronological order. If not extractable, return [].`;

   407→                cacheKey: `ohlcv-page:${symbol}:${days}`,
   408→              });
   409→              const data = parseJSON<OHLCV[]>(resp);

   411→                return data;
   412→              }
   413→            }
   414→          } catch {
   415→            continue;
   416→          }
   417→        }
   418→      }
   419→    }

   421→    // REMOVED: Strategy 3 (LLM fabrication of OHLCV) has been deleted.

   430→// ── Real News Fetching + LLM Sentiment ──────────────────────────
   431→export interface RealNewsArticle {
   432→  title: string;
   433→  url: string;
   434→  source: string;
   435→  snippet: string;
   436→  date: string;
   437→  sentiment: 'positive' | 'negative' | 'neutral';
   438→  sentimentScore: number;
   439→  positivePct: number;
   440→  negativePct: number;
   441→  neutralPct: number;
   442→}
   443→
   444→export async function fetchRealNews(symbol: string): Promise<{
   445→  articles: RealNewsArticle[];
   446→  summary: {
   447→    overallScore: number;
   448→    positivePct: number;
   449→    negativePct: number;
   450→    neutralPct: number;
   451→    positiveCount: number;
   452→    negativeCount: number;
   453→    neutralCount: number;
   454→    trending: string[];
   455→  };
   456→} | null> {
   457→  const cacheKey = `news:${symbol}`;
   458→  const cached = getCached<ReturnType<typeof fetchRealNews>>(cacheKey, 600_000);
   459→  if (cached) return cached;
   460→
   461→  try {
   462→    const companyName = getCompanyName(symbol);
   463→    const query = `${symbol} ${companyName} stock news today latest`;
   464→    const results = await webSearch(query, { num: 10, recencyDays: 7 });
   465→
   466→    if (!results.length) return null;
   467→
   468→    const articlesText = results.slice(0, 10)
   469→      .map((r, i) => `[${i + 1}] Title: ${r.name}\nSource: ${r.host_name}\nSnippet: ${r.snippet}\nDate: ${r.date || 'Recent'}`)
   470→      .join('\n\n');
   471→
   472→    const systemPrompt = `You are a financial sentiment analysis AI (FinBERT-class). Analyze the sentiment of each news article about ${symbol} stock.
   473→
   474→For EACH article, determine:
   475→- sentiment: "positive", "negative", or "neutral"
   476→- score: 0-100 (where >60 = positive, <40 = negative, 40-60 = neutral)
   477→- positivePct: confidence for positive sentiment (0-100)
   478→- negativePct: confidence for negative sentiment (0-100)
   479→- neutralPct: confidence for neutral sentiment (0-100)
   480→
   481→Also identify 4-6 trending topics/themes across all articles.
   482→
   483→Return ONLY a valid JSON object:
   484→{
   485→  "articles": [{ "index": 1, "sentiment": "positive", "score": 72, "positivePct": 65, "negativePct": 15, "neutralPct": 20 }],
   486→  "trending": ["Topic 1", "Topic 2"]
   487→}
   488→
   489→Analyze ALL ${Math.min(results.length, 10)} articles. Respond with JSON only.`;
   490→
   491→    const llmResponse = await llmChat(systemPrompt, `Analyze sentiment for these news articles about ${symbol}:\n\n${articlesText}`);
   492→
   493→    const parsed = parseJSON<{
   494→      articles: Array<{ index: number; sentiment: string; score: number; positivePct: number; negativePct: number; neutralPct: number }>;
   495→      trending: string[];
   496→    }>(llmResponse);
   497→
   498→    if (!parsed) return null;
   499→
   500→    const articles: RealNewsArticle[] = results.slice(0, 10).map((r, i) => {
   501→      const analysis = parsed.articles?.find((a) => a.index === i + 1);
   502→      return {
   503→        title: r.name,
   504→        url: r.url,
   505→        source: r.host_name,
   506→        snippet: r.snippet,
   507→        date: r.date || 'Recent',
   508→        sentiment: (analysis?.sentiment || 'neutral') as 'positive' | 'negative' | 'neutral',
   509→        sentimentScore: analysis?.score ?? 50,
   510→        positivePct: analysis?.positivePct ?? 33,
   511→        negativePct: analysis?.negativePct ?? 33,
   512→        neutralPct: analysis?.neutralPct ?? 34,
   513→      };
   514→    });
   515→
   516→    const positiveCount = articles.filter((a) => a.sentiment === 'positive').length;
   517→    const negativeCount = articles.filter((a) => a.sentiment === 'negative').length;
   518→    const neutralCount = articles.filter((a) => a.sentiment === 'neutral').length;
   519→    const avgPositive = articles.reduce((s, a) => s + a.positivePct, 0) / articles.length;
   520→    const avgNegative = articles.reduce((s, a) => s + a.negativePct, 0) / articles.length;
   521→    const avgNeutral = 100 - avgPositive - avgNegative;
   522→
   523→    const result = {
   524→      articles,
   525→      summary: {
   526→        overallScore: Math.round(50 + (avgPositive - avgNegative) * 0.5),
   527→        positivePct: Math.round(avgPositive),
   528→        negativePct: Math.round(avgNegative),
   529→        neutralPct: Math.round(Math.max(avgNeutral, 0)),
   530→        positiveCount,
   531→        negativeCount,
   532→        neutralCount,
   533→        trending: parsed.trending || [],
   534→      },
   535→    };
   536→
   537→    setCache(cacheKey, result);
   538→    return result;
   539→  } catch (error) {
   540→    console.error(`Failed to fetch real news for ${symbol}:`, error);
   541→    return null;
   542→  }
   543→}

   798→// ── LLM Forecast (real, no fake history) ─────────────────────────
   799→export async function fetchRealForecast(
   800→  symbol: string,
   801→  currentPrice: number,
   802→  timeframe: string,
   803→  indicators: Record<string, number | undefined>
   804→): Promise<{
   805→  points: Array<{ date: string; predicted: number; lower: number; upper: number; confidence: number }>;
   806→  summary: {
   807→    predictedPrice: number;
   808→    expectedChange: number;
   809→    volatility: number;
   810→    confidence: number;
   811→    analysis: string;
   812→  };
   813→} | null> {
   814→  const cacheKey = `forecast:${symbol}:${timeframe}:${currentPrice.toFixed(2)}`;
   815→  const cached = getCached<ReturnType<typeof fetchRealForecast>>(cacheKey, 900_000);
   816→  if (cached) return cached;
   817→
   818→  try {
   819→    const companyName = getCompanyName(symbol);
   820→    const days = timeframe === 'tomorrow' ? 1 : timeframe === '3days' ? 3 : timeframe === '7days' ? 7 : 30;
   821→
   822→    const systemPrompt = `You are an AI stock forecaster. Generate a realistic price forecast for ${symbol} (${companyName}).
   823→
   824→Current data:
   825→- Price: $${currentPrice.toFixed(2)}
   826→- RSI: ${indicators.rsi?.toFixed(1) || 'N/A'}
   827→- MACD: ${indicators.macd?.toFixed(2) || 'N/A'}
   828→- SMA(20): $${indicators.sma20?.toFixed(2) || 'N/A'}
   829→- SMA(50): $${indicators.sma50?.toFixed(2) || 'N/A'}
   830→- ATR: ${indicators.atr?.toFixed(2) || 'N/A'}
   831→
   832→Timeframe: Next ${days} trading day(s).
   833→
   834→Return ONLY valid JSON:
   835→{
   836→  "points": [
   837→    {"date": "YYYY-MM-DD", "predicted": <number>, "lower": <number>, "upper": <number>, "confidence": <number 40-90>}
   838→  ],
   839→  "summary": {
   840→    "predictedPrice": <number>,
   841→    "expectedChange": <percent number>,
   842→    "volatility": <number>,
   843→    "confidence": <number>,
   844→    "analysis": "<2-3 sentence analysis>"
   845→  }
   846→}
   847→
   848→Rules:
   849→- Generate exactly ${days} forecast point(s)
   850→- Predicted price must be within 5% of current price for short timeframes
   851→- Confidence intervals should widen with time
   852→- Use realistic volatility based on ATR
   853→- Dates must be future trading days (skip weekends)
   854→Respond with JSON only.`;
   855→
   856→    const llmResponse = await llmChat(systemPrompt, `Generate ${days}-day forecast for ${symbol}.`);
   857→    const parsed = parseJSON<Record<string, unknown>>(llmResponse);
   858→    if (!parsed) return null;
   859→
   860→    const points = Array.isArray(parsed.points)
   861→      ? (parsed.points as Array<Record<string, unknown>>).map((p) => ({
   862→          date: String(p.date || ''),
   863→          predicted: clamp(toNum(p.predicted, currentPrice), currentPrice * 0.9, currentPrice * 1.1),
   864→          lower: clamp(toNum(p.lower, currentPrice), currentPrice * 0.9, currentPrice),
   865→          upper: clamp(toNum(p.upper, currentPrice), currentPrice, currentPrice * 1.1),
   866→          confidence: clamp(toNum(p.confidence, 60), 30, 90),
   867→        }))
   868→      : [];
   869→
   870→    const summary = parsed.summary as Record<string, unknown> | undefined;
   871→    const predictedPrice = points.length > 0 ? points[points.length - 1].predicted : currentPrice;
   872→    const expectedChange = ((predictedPrice - currentPrice) / currentPrice) * 100;
   873→
   874→    const result = {
   875→      points,
   876→      summary: {
   877→        predictedPrice,
   878→        expectedChange: parseFloat(expectedChange.toFixed(2)),
   879→        volatility: toNum(summary?.volatility, 2),
   880→        confidence: clamp(toNum(summary?.confidence, 60), 30, 90),
   881→        analysis: String(summary?.analysis || 'No analysis available.'),
   882→      },
   883→    };
   884→
   885→    setCache(cacheKey, result);
   886→    return result;
   887→  } catch (error) {
   888→    console.error(`Failed to fetch forecast for ${symbol}:`, error);
   889→    return null;
   890→  }
