    89→// ─── Chart Tooltip ───────────────────────────────────────────────────
    90→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color?: string }>; label?: string }) {
    91→  if (!active || !payload?.length) return null;
    92→  return (
    93→    <div className="market-tooltip px-3 py-2 text-xs">
    94→      <p className="text-slate-400 mb-1">{label}</p>
    95→      {payload.map((entry, i) => (
    96→        <p key={i} style={{ color: entry.color || '#fff' }} className="font-semibold">
    97→          {entry.dataKey}: {typeof entry.value === 'number' ? entry.value.toFixed(2) : entry.value}
    98→        </p>
    99→      ))}
   100→    </div>
   101→  );
   102→}

   104→// ─── Derive indicator cards from API data ────────────────────────────
   105→function deriveIndicatorCards(ind: Record<string, number> | undefined, price: number): IndicatorCardData[] {
   106→  if (!ind) return [];
   107→  const cards: IndicatorCardData[] = [];
   108→
   109→  const rsi = ind.rsi;
   110→  if (rsi !== undefined && !isNaN(rsi)) {
   111→    cards.push({
   112→      label: 'RSI (14)',
   113→      value: rsi.toFixed(1),
   114→      raw: rsi,
   115→      signal: rsi > 70 ? 'bearish' : rsi < 30 ? 'bullish' : 'neutral',
   116→      description: rsi > 70 ? 'Overbought territory' : rsi < 30 ? 'Oversold territory' : 'Neutral zone',
   117→      icon: Activity,
   118→    });
   119→  }
   120→
   121→  const macd = ind.macd;
   122→  const macdSignal = ind.macdSignal;
   123→  const macdHist = ind.macdHistogram;
   124→  if (macd !== undefined && !isNaN(macd)) {
   125→    const signal = macdHist !== undefined && !isNaN(macdHist)
   126→      ? (macdHist > 0 ? 'bullish' : macdHist < 0 ? 'bearish' : 'neutral')
   127→      : 'neutral';
   128→    cards.push({
   129→      label: 'MACD',
   130→      value: macd.toFixed(3),
   131→      raw: macd,
   132→      signal,
   133→      description: macdSignal !== undefined ? `Signal: ${macdSignal.toFixed(3)}` : 'No signal line',
   134→      icon: Waves,
   135→    });
   136→  }
   137→
   138→  const sma20 = ind.sma20;
   139→  if (sma20 !== undefined && !isNaN(sma20)) {
   140→    cards.push({
   141→      label: 'SMA 20',
   142→      value: `$${sma20.toFixed(2)}`,
   143→      raw: price > 0 ? ((price - sma20) / sma20) * 100 : 0,
   144→      signal: price > sma20 ? 'bullish' : price < sma20 ? 'bearish' : 'neutral',
   145→      description: price > sma20 ? 'Price above SMA' : 'Price below SMA',
   146→      icon: TrendingUp,
   147→    });
   148→  }
   149→
   150→  const ema12 = ind.ema12;
   151→  if (ema12 !== undefined && !isNaN(ema12)) {
   152→    cards.push({
   153→      label: 'EMA 12',
   154→      value: `$${ema12.toFixed(2)}`,
   155→      raw: price > 0 ? ((price - ema12) / ema12) * 100 : 0,
   156→      signal: price > ema12 ? 'bullish' : price < ema12 ? 'bearish' : 'neutral',
   157→      description: 'Short-term exponential MA',
   158→      icon: Zap,
   159→    });
   160→  }
   161→
   162→  const bbUpper = ind.bbUpper;
   163→  const bbLower = ind.bbLower;
   164→  const bbMiddle = ind.bbMiddle;
   165→  if (bbUpper !== undefined && !isNaN(bbUpper) && bbLower !== undefined && !isNaN(bbLower)) {
   166→    const bbWidth = ((bbUpper - bbLower) / bbMiddle) * 100;
   167→    cards.push({
   168→      label: 'Bollinger Bands',
   169→      value: `${bbWidth.toFixed(1)}% width`,
   170→      raw: bbWidth,
   171→      signal: price >= bbUpper ? 'bearish' : price <= bbLower ? 'bullish' : 'neutral',
   172→      description: price >= bbUpper ? 'At upper band' : price <= bbLower ? 'At lower band' : 'Within bands',
   173→      icon: Layers,
   174→    });
   175→  }
   176→
   177→  const atr = ind.atr;
   178→  if (atr !== undefined && !isNaN(atr)) {
   179→    cards.push({
   180→      label: 'ATR (14)',
   181→      value: atr.toFixed(3),
   182→      raw: atr,
   183→      signal: 'neutral',
   184→      description: 'Average True Range',
   185→      icon: Target,
   186→    });
   187→  }
   188→
   189→  const vwap = ind.vwap;
   190→  if (vwap !== undefined && !isNaN(vwap)) {
   191→    cards.push({
   192→      label: 'VWAP',
   193→      value: `$${vwap.toFixed(2)}`,
   194→      raw: price > 0 ? ((price - vwap) / vwap) * 100 : 0,
   195→      signal: price > vwap ? 'bullish' : price < vwap ? 'bearish' : 'neutral',
   196→      description: price > vwap ? 'Above VWAP (bullish)' : 'Below VWAP (bearish)',
   197→      icon: BarChart3,
   198→    });
   199→  }
   200→
   201→  const stochK = ind.stochK;
   202→  const stochD = ind.stochD;
   203→  if (stochK !== undefined && !isNaN(stochK)) {
   204→    cards.push({
   205→      label: 'Stochastic RSI',
   206→      value: `K:${stochK.toFixed(1)} D:${stochD !== undefined && !isNaN(stochD) ? stochD.toFixed(1) : 'N/A'}`,
   207→      raw: stochK,
   208→      signal: stochK > 80 ? 'bearish' : stochK < 20 ? 'bullish' : 'neutral',
   209→      description: stochK > 80 ? 'Overbought' : stochK < 20 ? 'Oversold' : 'Neutral',
   210→      icon: Cpu,
   211→    });
   212→  }
   213→
   214→  const adx = ind.adx;
   215→  if (adx !== undefined && !isNaN(adx)) {
   216→    cards.push({
   217→      label: 'ADX (14)',
   218→      value: adx.toFixed(1),
   219→      raw: adx,
   220→      signal: adx > 25 ? 'bullish' : adx < 20 ? 'neutral' : 'neutral',
   221→      description: adx > 50 ? 'Strong trend' : adx > 25 ? 'Trending' : 'Weak trend / ranging',
   222→      icon: TrendingUp,
   223→    });
   224→  }
   225→
   226→  const cci = ind.cci;
   227→  if (cci !== undefined && !isNaN(cci)) {
   228→    cards.push({
   229→      label: 'CCI (20)',
   230→      value: cci.toFixed(1),
   231→      raw: cci,
   232→      signal: cci > 100 ? 'bearish' : cci < -100 ? 'bullish' : 'neutral',
   233→      description: cci > 100 ? 'Overbought' : cci < -100 ? 'Oversold' : 'Normal',
   234→      icon: Activity,
   235→    });
   236→  }
   237→
   238→  const obv = ind.obv;
   239→  if (obv !== undefined && !isNaN(obv)) {
   240→    cards.push({
   241→      label: 'OBV',
   242→      value: formatNumber(obv),
   243→      raw: 0,
   244→      signal: 'neutral',
   245→      description: 'On-Balance Volume',
   246→      icon: Volume2,
   247→    });
   248→  }
   249→
   250→  return cards;
   251→}

   253→// ─── Signal Summary Logic ────────────────────────────────────────────
   254→function getSignalSummary(cards: IndicatorCardData[]) {
   255→  let bullish = 0;
   256→  let bearish = 0;
   257→  let neutral = 0;
   258→  cards.forEach((c) => {
   259→    if (c.signal === 'bullish') bullish++;
   260→    else if (c.signal === 'bearish') bearish++;
   261→    else neutral++;
   262→  });
   263→  const total = bullish + bearish + neutral;
   264→  if (total === 0) return { label: 'No Data', bullish: 0, bearish: 0, neutral: 0, score: 0, color: '#64748b' };
   265→  const score = ((bullish - bearish) / total) * 100;
   266→  let label: string;
   267→  let color: string;
   268→  if (score > 25) { label = 'Strongly Bullish'; color = '#10b981'; }
   269→  else if (score > 10) { label = 'Bullish'; color = '#22c55e'; }
   270→  else if (score < -25) { label = 'Strongly Bearish'; color = '#ef4444'; }
   271→  else if (score < -10) { label = 'Bearish'; color = '#f97316'; }
   272→  else { label = 'Neutral'; color = '#eab308'; }
   273→  return { label, bullish, bearish, neutral, score, color };
   274→}

   276→// ─── Signal Badge Component ──────────────────────────────────────────
   277→function SignalBadge({ signal }: { signal: 'bullish' | 'bearish' | 'neutral' }) {
   278→  if (signal === 'bullish') return <Badge className="bg-emerald-500/15 text-emerald-400 text-[10px] border-0">BULLISH</Badge>;
   279→  if (signal === 'bearish') return <Badge className="bg-red-500/15 text-red-400 text-[10px] border-0">BEARISH</Badge>;
   280→  return <Badge className="bg-amber-500/15 text-amber-400 text-[10px] border-0">NEUTRAL</Badge>;
   281→}

   283→// ─── Main TechnicalAnalysisPage ─────────────────────────────────────
   284→export function TechnicalAnalysisPage() {
   285→  const { selectedStock } = useAppStore();
   286→  const [stockData, setStockData] = useState<StockResponse | null>(null);
   287→  const [loading, setLoading] = useState(true);
   288→  const [activeChart, setActiveChart] = useState('rsi');
   289→
   290→  const fetchData = useCallback(async () => {
   291→    setLoading(true);
   292→    try {
   293→      const res = await fetch(`/api/stock?symbol=${selectedStock.symbol}&days=90`);
   294→      const data: StockResponse = await res.json();
   295→      setStockData(data);
   296→    } catch {
   297→      setStockData(null);
   298→    } finally {
   299→      setLoading(false);
   300→    }
   301→  }, [selectedStock.symbol]);
   302→
   303→  useEffect(() => {
   304→    fetchData();
   305→  }, [fetchData]);
   306→
   307→  const ohlcv = stockData?.ohlcv || [];
   308→  const hasOHLCV = stockData?.hasOHLCV && ohlcv.length > 0;
   309→  const indicators = stockData?.indicators;
   310→  const series = stockData?.series;
   311→  const price = stockData?.stock?.price || 0;
   312→
   313→  const indicatorCards = deriveIndicatorCards(indicators, price);
   314→  const signalSummary = getSignalSummary(indicatorCards);
   315→
   316→  // Build chart data arrays aligned with OHLCV dates
   317→  const buildChartData = (keys: string[]) => {
   318→    if (!series || !hasOHLCV) return [];
   319→    return ohlcv.map((d, i) => {
   320→      const point: Record<string, string | number> = { date: d.date.slice(5) };
   321→      for (const key of keys) {
   322→        const arr = series[key];
   323→        if (!arr) continue;
   324→        // Series may be shorter (e.g. MACD) — align from end
   325→        const offset = ohlcv.length - arr.length;
   326→        const idx = i - offset;
   327→        if (idx >= 0 && idx < arr.length && !isNaN(arr[idx])) {
   328→          point[key] = arr[idx];
   329→        }
   330→      }
   331→      return point;
   332→    });
   333→  };
   334→
   335→  const rsiData = buildChartData(['rsi']);
   336→  const macdData = buildChartData(['macd', 'macdSignal', 'macdHistogram']);
   337→  const smaData = buildChartData(['sma20', 'sma50', 'ema12', 'close']);
   338→  const bbData = buildChartData(['bbUpper', 'bbMiddle', 'bbLower', 'close']);
   339→  const atrData = buildChartData(['atr']);
   340→  const vwapData = buildChartData(['vwap', 'close']);
   341→  const stochData = buildChartData(['stochK', 'stochD']);
   342→  const adxData = buildChartData(['adx', 'adxPlusDI', 'adxMinusDI']);
   343→  const cciData = buildChartData(['cci']);
   344→  const obvData = buildChartData(['obv']);
   345→
   346→  return (
   347→    <motion.div
   348→      variants={containerVariants}
   349→      initial="hidden"
   350→      animate="show"
   351→      className="space-y-6"
   352→    >
   353→      {/* ── Header ──────────────────────────────────────────────── */}
   354→      <motion.div variants={itemVariants}>
   355→        <SectionHeader
   356→          title="Technical Analysis"
   357→          subtitle={`${selectedStock.symbol} — 90 Day Indicators`}
   358→          badge={hasOHLCV ? { label: 'LIVE DATA', color: 'bg-emerald-500/15 text-emerald-400' } : undefined}
   359→          action={
   360→            <Button variant="outline" size="sm" onClick={fetchData} className="gradient-button border-0">
   361→              <RefreshCw className="w-3.5 h-3.5 mr-1.5" />
   362→              Refresh
   363→            </Button>
   364→          }
   365→        />
   366→      </motion.div>
   367→
   368→      {/* ── Signal Summary Bar ───────────────────────────────────── */}
   369→      {loading ? (
   370→        <motion.div variants={itemVariants} className="glass-card p-5">
   371→          <Skeleton className="h-16 w-full bg-white/5 rounded-xl" />
   372→        </motion.div>
   373→      ) : !hasOHLCV || indicatorCards.length === 0 ? (
   374→        <motion.div variants={itemVariants} className="glass-card p-8">
   375→          <EmptyState
   376→            icon={<Activity className="w-10 h-10" />}
   377→            title="No Live Data Available"
   378→            description={`No OHLCV data for ${selectedStock.symbol} to compute technical indicators. The data source may be temporarily unavailable.`}
   379→            action={
   380→              <Button variant="outline" size="sm" onClick={fetchData} className="gradient-button border-0">
   381→                <RefreshCw className="w-3.5 h-3.5 mr-2" />
   382→                Retry
   383→              </Button>
   384→            }
   385→          />
   386→        </motion.div>
   387→      ) : (
   388→        <>
   389→          <motion.div variants={itemVariants} className="glass-card p-5">
   390→            <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
   391→              <div className="flex items-center gap-4">
   392→                <CircularProgress

   409→                <div className="text-center">
   410→                  <div className="flex items-center gap-1.5 justify-center">
   411→                    <ArrowUpRight className="w-4 h-4 text-emerald-400" />
   412→                    <span className="text-lg font-bold text-emerald-400">{signalSummary.bullish}</span>
