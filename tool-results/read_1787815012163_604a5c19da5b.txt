     8→// ── In-memory Cache ──────────────────────────────────────────────
     9→interface CacheEntry<T> {
    10→  data: T;
    11→  timestamp: number;
    12→}
    13→
    14→const cache = new Map<string, CacheEntry<unknown>>();
    15→
    16→function getCached<T>(key: string, ttlMs: number): T | null {
    17→  const entry = cache.get(key);
    18→  if (entry && Date.now() - entry.timestamp < ttlMs) return entry.data as T;
    19→  cache.delete(key);
    20→  return null;
    21→}
    22→
    23→function setCache<T>(key: string, data: T): void {
    24→  if (cache.size > 300) {
    25→    const oldest = [...cache.entries()].sort((a, b) => a[1].timestamp - b[1].timestamp);
    26→    for (let i = 0; i < 80; i++) cache.delete(oldest[i][0]);
    27→  }
    28→  cache.set(key, { data, timestamp: Date.now() });
    29→}

    39→// ── Web Search ──────────────────────────────────────────────────
    40→export interface SearchResult {
    41→  url: string;
    42→  name: string;
    43→  snippet: string;
    44→  host_name: string;
    45→  rank: number;
    46→  date: string;
    47→  favicon: string;

    49→
    50→export async function webSearch(
    51→  query: string,
    52→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    53→): Promise<SearchResult[]> {
    54→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    55→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    56→
    57→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    58→  if (cached) return cached;
    59→
    60→  const zai = await getZAI();
    61→  const args: Record<string, unknown> = { query, num };
    62→  if (recencyDays) args.recency_days = recencyDays;
    63→
    64→  try {
    65→    const results = await Promise.race([
    66→      zai.functions.invoke('web_search', args),
    67→      new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    68→    ]);
    69→    const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    70→    setCache(cacheKey, typed);
    71→    return typed;
    72→  } catch (err) {
    73→    throw err;
    74→  }
    75→}

   105→// ── LLM Chat ────────────────────────────────────────────────────
   106→export async function llmChat(
   107→  systemPrompt: string,
   108→  userMessage: string,
   109→  options?: { cacheTtlMs?: number; cacheKey?: string }
   110→): Promise<string> {
   111→  const { cacheTtlMs = 600_000, cacheKey } = options || {};
   112→
   113→  if (cacheKey) {
   114→    const cached = getCached<string>(cacheKey, cacheTtlMs);
   115→    if (cached) return cached;
   116→  }
   117→
   118→  const zai = await getZAI();
   119→  const completion = await Promise.race([
   120→    zai.chat.completions.create({
   121→      messages: [
   122→        { role: 'assistant', content: systemPrompt },
   123→        { role: 'user', content: userMessage },
   124→      ],
   125→      thinking: { type: 'disabled' },
   126→    }),
   127→    new Promise((_, reject) => setTimeout(() => reject(new Error('LLM Timeout')), 20000)),
   128→  ]) as Awaited<ReturnType<typeof zai.chat.completions.create>>;
   129→
   130→
   131→  const response = completion.choices[0]?.message?.content || '';
   132→  if (cacheKey) setCache(cacheKey, response);
   133→  return response;
   134→}

   150→// ── Helpers ─────────────────────────────────────────────────────
   151→function toNum(val: unknown, fallback = 0): number {
   152→  if (typeof val === 'number') return val;
   153→  if (typeof val === 'string') {
   154→    const parsed = parseFloat(val.replace(/[^0-9.\-]/g, ''));
   155→    return isNaN(parsed) ? fallback : parsed;
   156→  }
   157→  return fallback;
   158→}

   161→  return Math.min(max, Math.max(min, val));

   170→  }

   392→// ── Real News Fetching + LLM Sentiment ──────────────────────────
   393→export interface RealNewsArticle {
   394→  title: string;

   406→export async function fetchRealNews(symbol: string): Promise<{
   407→  articles: RealNewsArticle[];
   408→  summary: {
   409→    overallScore: number;
   410→    positivePct: number;
   411→    negativePct: number;
   412→    neutralPct: number;
   413→    positiveCount: number;
   414→    negativeCount: number;
   415→    neutralCount: number;
   416→    trending: string[];
   417→  };
   418→} | null> {

   420→  const cached = getCached<ReturnType<typeof fetchRealNews>>(cacheKey, 600_000);
   421→  if (cached) return cached;

   423→  try {
   424→    const companyName = getCompanyName(symbol);
   425→    const query = `${symbol} ${companyName} stock news today latest`;
   426→    const results = await webSearch(query, { num: 10, recencyDays: 7 });
   427→
   428→    if (!results.length) return null;
   429→
   430→    const articlesText = results.slice(0, 10)
   431→      .map((r, i) => `[${i + 1}] Title: ${r.name}\nSource: ${r.host_name}\nSnippet: ${r.snippet}\nDate: ${r.date || 'Recent'}`)
   432→      .join('\n\n');
   433→
   434→    const systemPrompt = `You are a financial sentiment analysis AI (FinBERT-class). Analyze the sentiment of each news article about ${symbol} stock.
   435→
   436→For EACH article, determine:
   437→- sentiment: "positive", "negative", or "neutral"
   438→- score: 0-100 (where >60 = positive, <40 = negative, 40-60 = neutral)
   439→- positivePct: confidence for positive sentiment (0-100)
   440→- negativePct: confidence for negative sentiment (0-100)
   441→- neutralPct: confidence for neutral sentiment (0-100)
   442→
   443→Also identify 4-6 trending topics/themes across all articles.
   444→
   445→Return ONLY a valid JSON object:
   446→{
   447→  "articles": [{ "index": 1, "sentiment": "positive", "score": 72, "positivePct": 65, "negativePct": 15, "neutralPct": 20 }],
   448→  "trending": ["Topic 1", "Topic 2"]
   449→}
   450→
   451→Analyze ALL ${Math.min(results.length, 10)} articles. Respond with JSON only.`;
   452→
   453→    const llmResponse = await llmChat(systemPrompt, `Analyze sentiment for these news articles about ${symbol}:\n\n${articlesText}`);
   454→
   455→    const parsed = parseJSON<{
   456→      articles: Array<{ index: number; sentiment: string; score: number; positivePct: number; negativePct: number; neutralPct: number }>;
   457→      trending: string[];
   458→    }>(llmResponse);
   459→
   460→    if (!parsed) return null;
   461→
   462→    const articles: RealNewsArticle[] = results.slice(0, 10).map((r, i) => {
   463→      const analysis = parsed.articles?.find((a) => a.index === i + 1);
   464→      return {
   465→        title: r.name,
   466→        url: r.url,
   467→        source: r.host_name,
   468→        snippet: r.snippet,
   469→        date: r.date || 'Recent',
   470→        sentiment: (analysis?.sentiment || 'neutral') as 'positive' | 'negative' | 'neutral',
   471→        sentimentScore: analysis?.score ?? 50,
   472→        positivePct: analysis?.positivePct ?? 33,
   473→        negativePct: analysis?.negativePct ?? 33,
   474→        neutralPct: analysis?.neutralPct ?? 34,
   475→      };
   476→    });
   477→
   478→    const positiveCount = articles.filter((a) => a.sentiment === 'positive').length;
   479→    const negativeCount = articles.filter((a) => a.sentiment === 'negative').length;
   480→    const neutralCount = articles.filter((a) => a.sentiment === 'neutral').length;
   481→    const avgPositive = articles.reduce((s, a) => s + a.positivePct, 0) / articles.length;
   482→    const avgNegative = articles.reduce((s, a) => s + a.negativePct, 0) / articles.length;
   483→    const avgNeutral = 100 - avgPositive - avgNegative;
   484→
   485→    const result = {
   486→      articles,
   487→      summary: {
   488→        overallScore: Math.round(50 + (avgPositive - avgNegative) * 0.5),
   489→        positivePct: Math.round(avgPositive),
   490→        negativePct: Math.round(avgNegative),
   491→        neutralPct: Math.round(Math.max(avgNeutral, 0)),
   492→        positiveCount,
   493→        negativeCount,
   494→        neutralCount,
   495→        trending: parsed.trending || [],
   496→      },
   497→    };
   498→
   499→    setCache(cacheKey, result);
   500→    return result;
   501→  } catch (error) {
   502→    console.error(`Failed to fetch real news for ${symbol}:`, error);
   503→    return null;
   504→  }
   505→}

   761→export async function fetchRealForecast(
   762→  symbol: string,
   763→  currentPrice: number,
   764→  timeframe: string,
   765→  indicators: Record<string, number | undefined>
   766→): Promise<{
   767→  points: Array<{ date: string; predicted: number; lower: number; upper: number; confidence: number }>;
   768→  summary: {
   769→    predictedPrice: number;
   770→    expectedChange: number;
   771→    volatility: number;
   772→    confidence: number;
   773→    analysis: string;
   774→  };
   775→} | null> {
   776→  const cacheKey = `forecast:${symbol}:${timeframe}:${currentPrice.toFixed(2)}`;
   777→  const cached = getCached<ReturnType<typeof fetchRealForecast>>(cacheKey, 900_000);
   778→  if (cached) return cached;
   779→
   780→  try {
   781→    const companyName = getCompanyName(symbol);
   782→    const days = timeframe === 'tomorrow' ? 1 : timeframe === '3days' ? 3 : timeframe === '7days' ? 7 : 30;
   783→
   784→    const systemPrompt = `You are an AI stock forecaster. Generate a realistic price forecast for ${symbol} (${companyName}).
   785→
   786→Current data:
   787→- Price: $${currentPrice.toFixed(2)}
   788→- RSI: ${indicators.rsi?.toFixed(1) || 'N/A'}
   789→- MACD: ${indicators.macd?.toFixed(2) || 'N/A'}
   790→- SMA(20): $${indicators.sma20?.toFixed(2) || 'N/A'}
   791→- SMA(50): $${indicators.sma50?.toFixed(2) || 'N/A'}
   792→- ATR: ${indicators.atr?.toFixed(2) || 'N/A'}
   793→
   794→Timeframe: Next ${days} trading day(s).
   795→
   796→Return ONLY valid JSON:
   797→{
   798→  "points": [
   799→    {"date": "YYYY-MM-DD", "predicted": <number>, "lower": <number>, "upper": <number>, "confidence": <number 40-90>}
   800→  ],
   801→  "summary": {
   802→    "predictedPrice": <number>,
   803→    "expectedChange": <percent number>,
   804→    "volatility": <number>,
   805→    "confidence": <number>,
   806→    "analysis": "<2-3 sentence analysis>"
   807→  }
   808→}
   809→
   810→Rules:
   811→- Generate exactly ${days} forecast point(s)
   812→- Predicted price must be within 5% of current price for short timeframes
   813→- Confidence intervals should widen with time
   814→- Use realistic volatility based on ATR
   815→- Dates must be future trading days (skip weekends)
   816→Respond with JSON only.`;
   817→
   818→    const llmResponse = await llmChat(systemPrompt, `Generate ${days}-day forecast for ${symbol}.`);
   819→    const parsed = parseJSON<Record<string, unknown>>(llmResponse);
   820→    if (!parsed) return null;
   821→
   822→    const points = Array.isArray(parsed.points)
   823→      ? (parsed.points as Array<Record<string, unknown>>).map((p) => ({
   824→          date: String(p.date || ''),
   825→          predicted: clamp(toNum(p.predicted, currentPrice), currentPrice * 0.9, currentPrice * 1.1),
   826→          lower: clamp(toNum(p.lower, currentPrice), currentPrice * 0.9, currentPrice),
   827→          upper: clamp(toNum(p.upper, currentPrice), currentPrice, currentPrice * 1.1),
   828→          confidence: clamp(toNum(p.confidence, 60), 30, 90),
   829→        }))
   830→      : [];
   831→
   832→    const summary = parsed.summary as Record<string, unknown> | undefined;
   833→    const predictedPrice = points.length > 0 ? points[points.length - 1].predicted : currentPrice;
   834→    const expectedChange = ((predictedPrice - currentPrice) / currentPrice) * 100;
   835→
   836→    const result = {
   837→      points,
   838→      summary: {
   839→        predictedPrice,
   840→        expectedChange: parseFloat(expectedChange.toFixed(2)),
   841→        volatility: toNum(summary?.volatility, 2),
   842→        confidence: clamp(toNum(summary?.confidence, 60), 30, 90),
   843→        analysis: String(summary?.analysis || 'No analysis available.'),
   844→      },
   845→    };
   846→
   847→    setCache(cacheKey, result);
   848→    return result;
   849→  } catch (error) {
   850→    console.error(`Failed to fetch forecast for ${symbol}:`, error);
   851→    return null;
   852→  }
