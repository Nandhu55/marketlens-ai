   102→// ─── Chart Tooltip ───────────────────────────────────────────────────
   103→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string }>; label?: string }) {
   104→  if (!active || !payload?.length) return null;
   105→  return (
   106→    <div className="market-tooltip px-3 py-2 text-xs">
   107→      <p className="text-slate-400 mb-1">{label}</p>
   108→      <p className="text-white font-semibold">${payload[0].value.toFixed(2)}</p>
   109→    </div>
   110→  );
   111→}

   113→// ─── Fear & Greed Semicircle Gauge ───────────────────────────────────
   114→function FearGreedGauge({ value }: { value: number }) {
   115→  if (value <= 0) {
   116→    return (
   117→      <div className="flex flex-col items-center justify-center h-full">
   118→        <p className="text-slate-500 text-sm">No Data</p>
   119→      </div>
   120→    );
   121→  }
   122→
   123→  const clamped = Math.max(0, Math.min(100, value));
   124→
   125→  const getZone = () => {
   126→    if (clamped <= 25) return { label: 'Extreme Fear', color: '#ef4444' };
   127→    if (clamped <= 45) return { label: 'Fear', color: '#f97316' };
   128→    if (clamped <= 55) return { label: 'Neutral', color: '#eab308' };
   129→    if (clamped <= 75) return { label: 'Greed', color: '#22c55e' };
   130→    return { label: 'Extreme Greed', color: '#10b981' };
   131→  };
   132→
   133→  const zone = getZone();
   134→  const width = 260;
   135→  const height = 145;
   136→  const cx = width / 2;
   137→  const cy = height - 10;
   138→  const radius = 100;
   139→  const strokeWidth = 18;
   140→  const valueAngle = Math.PI - (clamped / 100) * Math.PI;
   141→  const needleLen = radius - strokeWidth - 5;
   142→  const needleTipX = cx - needleLen * Math.cos(valueAngle);
   143→  const needleTipY = cy - needleLen * Math.sin(valueAngle);
   144→  const startX = cx - radius;
   145→  const endX = cx + radius;
   146→
   147→  const segments = [
   148→    { from: 0, to: 25, color: '#ef4444' },
   149→    { from: 25, to: 45, color: '#f97316' },
   150→    { from: 45, to: 55, color: '#eab308' },
   151→    { from: 55, to: 75, color: '#22c55e' },
   152→    { from: 75, to: 100, color: '#10b981' },
   153→  ];
   154→
   155→  const getArcPath = (fromPct: number, toPct: number) => {
   156→    const startAngle = Math.PI - (fromPct / 100) * Math.PI;
   157→    const endAngle = Math.PI - (toPct / 100) * Math.PI;
   158→    const sx = cx + radius * Math.cos(startAngle);
   159→    const sy = cy - radius * Math.sin(startAngle);
   160→    const ex = cx + radius * Math.cos(endAngle);
   161→    const ey = cy - radius * Math.sin(endAngle);
   162→    const largeArc = (fromPct - toPct) > 50 ? 1 : 0;
   163→    return `M ${sx} ${sy} A ${radius} ${radius} 0 ${largeArc} 0 ${ex} ${ey}`;
   164→  };
   165→
   166→  return (
   167→    <div className="flex flex-col items-center">
   168→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
   169→        {segments.map((seg) => (
   170→          <path
   171→            key={`${seg.from}-${seg.to}`}
   172→            d={getArcPath(seg.from, seg.to)}
   173→            fill="none"
   174→            stroke={seg.color}
   175→            strokeWidth={strokeWidth}
   176→            strokeLinecap="butt"
   177→            opacity={0.2}
   178→          />
   179→        ))}
   180→        <motion.path
   181→          d={getArcPath(0, clamped)}
   182→          fill="none"
   183→          stroke={zone.color}
   184→          strokeWidth={strokeWidth}
   185→          strokeLinecap="round"
   186→          initial={{ pathLength: 0 }}
   187→          animate={{ pathLength: 1 }}
   188→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   189→          style={{ filter: `drop-shadow(0 0 8px ${zone.color}60)` }}
   190→        />
   191→        <motion.line
   192→          x1={cx} y1={cy} x2={needleTipX} y2={needleTipY}
   193→          stroke={zone.color} strokeWidth={2.5} strokeLinecap="round"
   194→          initial={{ opacity: 0 }}
   195→          animate={{ opacity: 1, x2: needleTipX, y2: needleTipY }}
   196→          transition={{ duration: 1, delay: 0.5 }}
   197→          style={{ filter: `drop-shadow(0 0 4px ${zone.color}80)` }}
   198→        />
   199→        <circle cx={cx} cy={cy} r={5} fill={zone.color} />
   200→        <text x={startX - 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">0</text>
   201→        <text x={endX + 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">100</text>
   202→      </svg>
   203→      <div className="text-center -mt-1">
   204→        <div className="text-3xl font-bold" style={{ color: zone.color }}>
   205→          {Math.round(clamped)}
   206→        </div>
   207→        <div className="text-xs font-medium mt-0.5" style={{ color: zone.color }}>
   208→          {zone.label}
   209→        </div>
   210→        <div className="text-[10px] text-slate-500 mt-1 uppercase tracking-widest">Fear &amp; Greed Index</div>
   211→      </div>
   212→    </div>
   213→  );
   214→}

   217→function TickerItem({ symbol, price, changePercent }: { symbol: string; price: number; changePercent: number }) {
   218→  const isPositive = changePercent >= 0;
   219→  return (
   220→    <div className="flex items-center gap-2.5 px-5 py-2 whitespace-nowrap">
   221→      <span className="text-xs font-bold text-white tracking-wider">{symbol}</span>
   222→      <span className="text-xs text-slate-300">${formatPrice(price)}</span>
   223→      <span
   224→        className={cn(
   225→          'text-xs font-semibold flex items-center gap-0.5',
   226→          isPositive ? 'text-emerald-400' : 'text-red-400'
   227→        )}
   228→      >
   229→        {isPositive ? <ArrowUpRight className="w-3 h-3" /> : <ArrowDownRight className="w-3 h-3" />}
   230→        {formatPercent(changePercent)}
   231→      </span>
   232→    </div>
   233→  );
   234→}

   250→// ─── Main Dashboard Page ────────────────────────────────────────────
   251→export function DashboardPage() {
   252→  const { realQuotes, selectedStock, watchlist, isLive } = useAppStore();
   253→  const [overviewData, setOverviewData] = useState<MarketOverviewResponse | null>(null);
   254→  const [stockData, setStockData] = useState<StockDataResponse | null>(null);
   255→  const [loading, setLoading] = useState(true);
   256→  const [stockLoading, setStockLoading] = useState(true);
   257→  const [error, setError] = useState(false);
   258→
   259→  const fetchMarketOverview = useCallback(async () => {
   260→    setLoading(true);
   261→    setError(false);
   262→    try {
   263→      const res = await fetch('/api/market-overview');
   264→      const data: MarketOverviewResponse = await res.json();
   265→      setOverviewData(data);
   266→    } catch {
   267→      setError(true);
   268→    } finally {
   269→      setLoading(false);
   270→    }
   271→  }, []);
   272→
   273→  const fetchStockData = useCallback(async () => {
   274→    setStockLoading(true);
   275→    try {
   276→      const res = await fetch(`/api/stock?symbol=${selectedStock.symbol}&days=30`);
   277→      const data: StockDataResponse = await res.json();
   278→      setStockData(data);
   279→    } catch {
   280→      setStockData(null);
   281→    } finally {
   282→      setStockLoading(false);
   283→    }
   284→  }, [selectedStock.symbol]);
   285→
   286→  useEffect(() => {
   287→    fetchMarketOverview();
   288→  }, [fetchMarketOverview]);
   289→
   290→  useEffect(() => {
   291→    fetchStockData();
   292→  }, [fetchStockData]);
   293→
   294→  // Ticker data from realQuotes
   295→  const tickerItems = Object.entries(realQuotes).map(([symbol, q]) => ({
   296→    symbol,
   297→    price: q.price,
   298→    changePercent: q.changePercent,
   299→  }));
   300→
   301→  // Chart data from stock OHLCV
   302→  const chartData = stockData?.ohlcv?.map((d) => ({
   303→    date: d.date.slice(5),
   304→    price: d.close,
   305→  })) || [];
   306→
   307→  // Aggregated values from overview
   308→  const totalVolume = overviewData?.quotes
   309→    ? Object.values(overviewData.quotes).reduce((sum, q) => sum + (q.volume || 0), 0)
   310→    : 0;
   311→
   312→  const avgChange = overviewData?.quotes
   313→    ? (() => {
   314→        const vals = Object.values(overviewData.quotes).map((q) => q.changePercent);
   315→        return vals.length > 0 ? vals.reduce((a, b) => a + b, 0) / vals.length : 0;
   316→      })()
   317→    : 0;
   318→
   319→  const topGainer = overviewData?.overview?.topGainers?.[0];
   320→  const topLoser = overviewData?.overview?.topLosers?.[0];
   321→
   322→  const hasRealQuotes = tickerItems.length > 0;
   323→  const hasOverview = !!overviewData?.hasRealData;
   324→  const fearGreedValue = overviewData?.overview?.fearGreedIndex || 0;
   325→
   326→  return (
   327→    <motion.div
   328→      variants={containerVariants}
   329→      initial="hidden"
   330→      animate="show"
   331→      className="space-y-6"
   332→    >
   333→      {/* ── Market Ticker Bar ─────────────────────────────────── */}
   334→      {hasRealQuotes ? (
   335→        <motion.div variants={itemVariants} className="glass-card overflow-hidden">
   336→          <div className="flex items-center">
   337→            <div className="flex-shrink-0 px-4 py-2 border-r border-white/5 flex items-center gap-2">
   338→              {isLive && (
   339→                <span className="flex items-center gap-1.5 text-[10px] font-semibold text-emerald-400 uppercase tracking-wider">
   340→                  <Zap className="w-3 h-3" />
   341→                  Live
   342→                </span>
   343→              )}
   344→              <CandlestickChart className="w-3.5 h-3.5 text-indigo-400" />
   345→            </div>
   346→            <div className="overflow-hidden flex-1">
   347→              <div className="ticker-scroll flex">
   348→                {[...tickerItems, ...tickerItems].map((item, i) => (
   349→                  <TickerItem key={`${item.symbol}-${i}`} {...item} />
   350→                ))}
   351→              </div>
   352→            </div>
   353→          </div>
   354→        </motion.div>
   355→      ) : (
   356→        <motion.div variants={itemVariants} className="glass-card overflow-hidden">
   357→          <div className="flex items-center justify-center px-4 py-3">
   358→            <p className="text-sm text-slate-500">Waiting for live market data...</p>
   359→          </div>
   360→        </motion.div>
   361→      )}
   362→
   363→      {/* ── KPI Cards ──────────────────────────────────────────── */}
   364→      <motion.div variants={itemVariants}>
   365→        {loading ? (
   366→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   367→            {Array.from({ length: 8 }).map((_, i) => (
   368→              <KPISkeleton key={i} />
   369→            ))}
   370→          </div>
   371→        ) : error ? (
   372→          <div className="flex flex-col items-center justify-center py-12">
   373→            <EmptyState
   374→              icon={<Activity className="w-10 h-10" />}
   375→              title="No Live Data Available"

   389→              value={totalVolume > 0 ? formatNumber(totalVolume) : 'No Data'}
   390→              icon={BarChart3}

   392→              glowColor="glow-cyan"
   393→              index={0}
   394→            />
   395→            <KPICard

   398→              change={overviewData && totalVolume > 0 ? formatPercent(avgChange) : undefined}
   399→              changeType={avgChange >= 0 ? 'positive' : 'negative'}
   400→              icon={TrendingUp}
   401→              iconColor="text-emerald-400"
   402→              glowColor="glow-green"
   403→              index={1}
   404→            />
   405→            <KPICard
   406→              title="Top Gainer"
   407→              value={topGainer ? `${topGainer.symbol}` : 'No Data'}
   408→              change={topGainer ? `+${topGainer.change.toFixed(2)}%` : undefined}
   409→              changeType="positive"
   410→              icon={ArrowUpRight}
   411→              iconColor="text-emerald-400"
   412→              glowColor="glow-green"

   415→            <KPICard

   428→              icon={Target}
   429→              iconColor="text-violet-400"
   430→              glowColor="glow-purple"
   431→              index={4}
   432→            />
   433→            <KPICard

   436→              icon={TrendingUp}
   437→              iconColor={avgChange >= 0 ? 'text-emerald-400' : 'text-red-400'}
   438→              glowColor={avgChange >= 0 ? 'glow-green' : 'glow-red'}
   439→              index={5}
   440→            />

   446→              glowColor="glow-cyan"
   447→              index={6}
