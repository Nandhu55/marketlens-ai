    37→    37→export async function webSearch(
    38→    38→  query: string,
    39→    39→  options?: { num?: number; recencyDays?: number; cacheTtlMs?: number }
    40→    40→): Promise<SearchResult[]> {
    41→    41→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    42→    42→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;
    43→    43→
    44→    44→  const cached = getCached<SearchResult[]>(cacheKey, cacheTtlMs);
    45→    45→  if (cached) return cached;
    46→    46→
    47→    47→  const zai = await getZAI();
    48→    48→  const args: Record<string, unknown> = { query, num };
    49→    49→  if (recencyDays) args.recency_days = recencyDays;
    50→    50→
    51→    51→  try {
    52→    52→    const results = await Promise.race([
    53→    53→      zai.functions.invoke('web_search', args),
    54→    54→      new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    55→    55→    ]);
    56→    56→    const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    57→    57→    setCache(cacheKey, typed);
    58→    58→    return typed;
    59→    59→  } catch (err) {
    60→    60→    throw err;
    61→    61→  }
    62→    62→}

   138→   138→function toNum(val: unknown, fallback = 0): number {
   139→   139→  if (typeof val === 'number') return val;
   140→   140→  if (typeof val === 'string') {
   141→   141→    const parsed = parseFloat(val.replace(/[^0-9.\-]/g, ''));
   142→   142→    return isNaN(parsed) ? fallback : parsed;
   143→   143→  }
   144→   144→  return fallback;
   145→   145→}

   336→   336→export interface RealNewsArticle {
   337→   337→  title: string;
   338→   338→  url: string;
   339→   339→  source: string;

   349→   349→export async function fetchRealNews(symbol: string): Promise<{
   350→   350→  articles: RealNewsArticle[];
   351→   351→  summary: {
   352→   352→    overallScore: number;
   353→   353→    positivePct: number;
   354→   354→    negativePct: number;
   355→   355→    neutralPct: number;
   356→   356→    positiveCount: number;
   357→   357→    negativeCount: number;
   358→   358→    neutralCount: number;
   359→   359→    trending: string[];
   360→   360→  };
   361→   361→} | null> {
   362→   362→  const cacheKey = `news:${symbol}`;
   363→   363→  const cached = getCached<ReturnType<typeof fetchRealNews>>(cacheKey, 600_000);
   364→   364→  if (cached) return cached;
   365→   365→
   366→   366→  try {
   367→   367→    const companyName = getCompanyName(symbol);
   368→   368→    const query = `${symbol} ${companyName} stock news today latest`;
   369→   369→    const results = await webSearch(query, { num: 10, recencyDays: 7 });
   370→   370→
   371→   371→    if (!results.length) return null;
   372→   372→
   373→   373→    const articlesText = results.slice(0, 10)
   374→   374→      .map((r, i) => `[${i + 1}] Title: ${r.name}\nSource: ${r.host_name}\nSnippet: ${r.snippet}\nDate: ${r.date || 'Recent'}`)
   375→   375→      .join('\n\n');
   376→   376→
   377→   377→    const systemPrompt = `You are a financial sentiment analysis AI (FinBERT-class). Analyze the sentiment of each news article about ${symbol} stock.
   378→   378→
   379→   379→For EACH article, determine:
   380→   380→- sentiment: "positive", "negative", or "neutral"
   381→   381→- score: 0-100 (where >60 = positive, <40 = negative, 40-60 = neutral)
   382→   382→- positivePct: confidence for positive sentiment (0-100)
   383→   383→- negativePct: confidence for negative sentiment (0-100)
   384→   384→- neutralPct: confidence for neutral sentiment (0-100)
   385→   385→
   386→   386→Also identify 4-6 trending topics/themes across all articles.
   387→   387→
   388→   388→Return ONLY a valid JSON object:
   389→   389→{
   390→   390→  "articles": [{ "index": 1, "sentiment": "positive", "score": 72, "positivePct": 65, "negativePct": 15, "neutralPct": 20 }],
   391→   391→  "trending": ["Topic 1", "Topic 2"]
   392→   392→}
   393→   393→
   394→   394→Analyze ALL ${Math.min(results.length, 10)} articles. Respond with JSON only.`;
   395→   395→
   396→   396→    const llmResponse = await llmChat(systemPrompt, `Analyze sentiment for these news articles about ${symbol}:\n\n${articlesText}`);
   397→   397→
   398→   398→    const parsed = parseJSON<{
   399→   399→      articles: Array<{ index: number; sentiment: string; score: number; positivePct: number; negativePct: number; neutralPct: number }>;
   400→   400→      trending: string[];
   401→   401→    }>(llmResponse);
   402→   402→
   403→   403→    if (!parsed) return null;
   404→   404→
   405→   405→    const articles: RealNewsArticle[] = results.slice(0, 10).map((r, i) => {
   406→   406→      const analysis = parsed.articles?.find((a) => a.index === i + 1);
   407→   407→      return {
   408→   408→        title: r.name,
   409→   409→        url: r.url,
   410→   410→        source: r.host_name,
   411→   411→        snippet: r.snippet,
   412→   412→        date: r.date || 'Recent',
   413→   413→        sentiment: (analysis?.sentiment || 'neutral') as 'positive' | 'negative' | 'neutral',
   414→   414→        sentimentScore: analysis?.score ?? 50,
   415→   415→        positivePct: analysis?.positivePct ?? 33,
   416→   416→        negativePct: analysis?.negativePct ?? 33,
   417→   417→        neutralPct: analysis?.neutralPct ?? 34,
   418→   418→      };
   419→   419→    });
   420→   420→
   421→   421→    const positiveCount = articles.filter((a) => a.sentiment === 'positive').length;
   422→   422→    const negativeCount = articles.filter((a) => a.sentiment === 'negative').length;
   423→   423→    const neutralCount = articles.filter((a) => a.sentiment === 'neutral').length;
   424→   424→    const avgPositive = articles.reduce((s, a) => s + a.positivePct, 0) / articles.length;
   425→   425→    const avgNegative = articles.reduce((s, a) => s + a.negativePct, 0) / articles.length;
   426→   426→    const avgNeutral = 100 - avgPositive - avgNegative;
   427→   427→
   428→   428→    const result = {
   429→   429→      articles,
   430→   430→      summary: {
   431→   431→        overallScore: Math.round(50 + (avgPositive - avgNegative) * 0.5),
   432→   432→        positivePct: Math.round(avgPositive),
   433→   433→        negativePct: Math.round(avgNegative),
   434→   434→        neutralPct: Math.round(Math.max(avgNeutral, 0)),
   435→   435→        positiveCount,
   436→   436→        negativeCount,
   437→   437→        neutralCount,
   438→   438→        trending: parsed.trending || [],
   439→   439→      },
   440→   440→    };
   441→   441→
   442→   442→    setCache(cacheKey, result);
   443→   443→    return result;
   444→   444→  } catch (error) {
   445→   445→    console.error(`Failed to fetch real news for ${symbol}:`, error);
   446→   446→    return null;
   447→   447→  }
