     9→     9→interface CacheEntry<T> {
    10→    10→  data: T;
    11→    11→  timestamp: number;
    12→    12→}
    13→    13→
    14→    14→const cache = new Map<string, CacheEntry<unknown>>();
    15→    15→
    16→    16→function getCached<T>(key: string, ttlMs: number): T | null {
    17→    17→  const entry = cache.get(key);
    18→    18→  if (entry && Date.now() - entry.timestamp < ttlMs) return entry.data as T;
    19→    19→  cache.delete(key);
    20→    20→  return null;
    21→    21→}
    22→    22→
    23→    23→function setCache<T>(key: string, data: T): void {
    24→    24→  if (cache.size > 300) {
    25→    25→    const oldest = [...cache.entries()].sort((a, b) => a[1].timestamp - b[1].timestamp);
    26→    26→    for (let i = 0; i < 80; i++) cache.delete(oldest[i][0]);
    27→    27→  }
    28→    28→  cache.set(key, { data, timestamp: Date.now() });
    29→    29→}

    53→    53→): Promise<SearchResult[]> {
    54→    54→  const { num = 8, recencyDays, cacheTtlMs = 300_000 } = options || {};
    55→    55→  const cacheKey = `search:${query}:${num}:${recencyDays || 'all'}`;

    59→    59→

    61→    61→  const args: Record<string, unknown> = { query, num };
    62→    62→  if (recencyDays) args.recency_days = recencyDays;
    63→    63→
    64→    64→  try {
    65→    65→    const results = await Promise.race([
    66→    66→      zai.functions.invoke('web_search', args),
    67→    67→      new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), 15000)),
    68→    68→    ]);
    69→    69→    const typed = (results as SearchResult[]).map((r, i) => ({ ...r, rank: r.rank ?? i + 1 }));
    70→    70→    setCache(cacheKey, typed);
    71→    71→    return typed;
    72→    72→  } catch (err) {
    73→    73→    throw err;
    74→    74→  }
    75→    75→}
