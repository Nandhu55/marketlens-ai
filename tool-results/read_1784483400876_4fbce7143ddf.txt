   246→function generateArticles(symbol: string): NewsArticle[] {
   247→  const headlines = HEADLINES_BY_STOCK[symbol] || HEADLINES_BY_STOCK.AAPL;
   248→  const stock = STOCKS[symbol] || STOCKS.AAPL;
   249→  const hash = symbol.split('').reduce((a, c) => a + c.charCodeAt(0), 0);
   250→
   251→  return headlines.map((headline, i) => {
   252→    const r1 = seededRandom(hash * 100 + i * 7);
   253→    const r2 = seededRandom(hash * 200 + i * 13);
   254→    const r3 = seededRandom(hash * 300 + i * 17);
   255→    const r4 = seededRandom(hash * 400 + i * 23);
   256→
   257→    let sentiment: Sentiment;
   258→    if (r1 > 0.55) sentiment = 'positive';
   259→    else if (r1 > 0.25) sentiment = 'neutral';
   260→    else sentiment = 'negative';
   261→
   262→    const positivePercent = sentiment === 'positive' ? 55 + Math.round(r2 * 30) : sentiment === 'negative' ? 10 + Math.round(r2 * 20) : 20 + Math.round(r2 * 20);
   263→    const negativePercent = sentiment === 'negative' ? 50 + Math.round(r3 * 30) : sentiment === 'positive' ? 5 + Math.round(r3 * 15) : 15 + Math.round(r3 * 15);
   264→    const neutralPercent = 100 - positivePercent - negativePercent;
   265→
   266→    const times = ['15m ago', '32m ago', '1h ago', '2h ago', '3h ago', '5h ago', '7h ago', '8h ago', '12h ago', '18h ago', '1d ago', '2d ago'];
   267→
   268→    return {
   269→      id: i,
   270→      headline,
   271→      source: SOURCES[Math.floor(r4 * SOURCES.length)],
   272→      time: times[i % times.length],
   273→      sentiment,
   274→      positivePercent,
   275→      negativePercent,
   276→      neutralPercent,
   277→      summary: `${stock.name} related development. Market impact assessment pending detailed analysis.`,
   278→    };
   279→  });
   280→}
