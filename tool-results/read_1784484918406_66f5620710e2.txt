   110→   110→// ─── Custom Candlestick Shape ────────────────────────────────────────
   111→   111→function CandlestickBar(props: any) {
   112→   112→  const { x, y, width, height, payload } = props;
   113→   113→  if (!payload || isNaN(payload.body[0]) || isNaN(payload.body[1])) return null;
   114→   114→
   115→   115→  const open = payload.open;
   116→   116→  const close = payload.close;
   117→   117→  const low = payload.low;
   118→   118→  const high = payload.high;
   119→   119→  const isBull = close >= open;
   120→   120→  const color = isBull ? CHART_COLORS.bull : CHART_COLORS.bear;
   121→   121→
   122→   122→  const barWidth = Math.max(width * 0.6, 3);
   123→   123→  const cx = x + width / 2;
   124→   124→  const bodyTop = y;
   125→   125→  const bodyBottom = y + height;
   126→   126→  const bodyHeight = Math.max(height, 1);
   127→   127→
   128→   128→  // Calculate y-axis scale from the chart dimensions
   129→   129→  const priceRange = high - low;
   130→   130→  const totalHeight = props.backgroundHeight || 300;
   131→   131→  const yScale = totalHeight / ((props.yAxisMax || high) - (props.yAxisMin || low));
   132→   132→
   133→   133→  const highY = bodyTop - (high - Math.max(open, close)) * yScale;
   134→   134→  const lowY = bodyBottom + (Math.min(open, close) - low) * yScale;
   135→   135→
   136→   136→  return (
   137→   137→    <g>
   138→   138→      {/* Wick */}
   139→   139→      <line
   140→   140→        x1={cx}
   141→   141→        y1={lowY}
   142→   142→        x2={cx}
   143→   143→        y2={highY}
   144→   144→        stroke={color}
   145→   145→        strokeWidth={1}
   146→   146→        opacity={0.7}
   147→   147→      />
   148→   148→      {/* Body */}
   149→   149→      <rect
   150→   150→        x={cx - barWidth / 2}
   151→   151→        y={bodyTop}
   152→   152→        width={barWidth}
   153→   153→        height={bodyHeight}
   154→   154→        fill={isBull ? color : color}
   155→   155→        opacity={isBull ? 0.9 : 0.9}
   156→   156→        stroke={color}
   157→   157→        strokeWidth={0.5}
   158→   158→        rx={1}
   159→   159→      />
   160→   160→    </g>
   161→   161→  );
   162→   162→}
