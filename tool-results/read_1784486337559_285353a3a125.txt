    44→    44→// ─── Seeded PRNG based on symbol ──────────────────────────────────────────
    45→    45→function seededRandom(seed: number): () => number {
    46→    46→  let s = seed;
    47→    47→  return () => {
    48→    48→    s = (s * 16807 + 0) % 2147483647;
    49→    49→    return (s - 1) / 2147483646;
    50→    50→  };
    51→    51→}
    52→    52→
    53→    53→function symbolSeed(symbol: string): number {
    54→    54→  return symbol.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0) + 42;
    55→    55→}

    75→    75→function generatePrediction(symbol: string, currentPrice: number): PredictionData {
    76→    76→  const rng = seededRandom(symbolSeed(symbol));
    77→    77→
    78→    78→  const bullishBase = 35 + rng() * 40;
    79→    79→  const bullishProb = parseFloat(bullishBase.toFixed(1));
    80→    80→  const bearishProb = parseFloat((100 - bullishProb).toFixed(1));
    81→    81→
    82→    82→  const volatilityScore = parseFloat((20 + rng() * 60).toFixed(1));
    83→    83→  const confidenceScore = parseFloat((65 + rng() * 30).toFixed(1));
    84→    84→
    85→    85→  const spread = currentPrice * (0.02 + rng() * 0.06);
    86→    86→  const upperBound = parseFloat((currentPrice + spread * (0.4 + rng() * 0.6)).toFixed(2));
    87→    87→  const lowerBound = parseFloat((currentPrice - spread * (0.4 + rng() * 0.6)).toFixed(2));
    88→    88→
    89→    89→  let recommendation: 'BUY' | 'HOLD' | 'SELL';
    90→    90→  if (bullishProb > 62) recommendation = 'BUY';
    91→    91→  else if (bullishProb < 38) recommendation = 'SELL';
    92→    92→  else recommendation = 'HOLD';
    93→    93→
    94→    94→  let riskLevel: 'Low' | 'Medium' | 'High' | 'Extreme';
    95→    95→  if (volatilityScore < 35) riskLevel = 'Low';
    96→    96→  else if (volatilityScore < 55) riskLevel = 'Medium';
    97→    97→  else if (volatilityScore < 75) riskLevel = 'High';
    98→    98→  else riskLevel = 'Extreme';
    99→    99→
   100→   100→  return {
   101→   101→    bullishProb,
   102→   102→    bearishProb,
   103→   103→    volatilityScore,
   104→   104→    confidenceScore,
   105→   105→    upperBound,
   106→   106→    lowerBound,
   107→   107→    recommendation,
   108→   108→    riskLevel,
   109→   109→    accuracy: parseFloat((82 + rng() * 12).toFixed(1)),
   110→   110→    precision: parseFloat((80 + rng() * 12).toFixed(1)),
   111→   111→    recall: parseFloat((83 + rng() * 12).toFixed(1)),
   112→   112→    f1Score: parseFloat((81 + rng() * 12).toFixed(1)),
   113→   113→    rmse: parseFloat((1.5 + rng() * 2.5).toFixed(2)),
   114→   114→    mae: parseFloat((1.0 + rng() * 2.0).toFixed(2)),
   115→   115→  };
   116→   116→}

   118→   118→// ─── Historical accuracy chart data ──────────────────────────────────────
   119→   119→interface AccuracyPoint {
   120→   120→  date: string;
   121→   121→  predicted: number;
   122→   122→  actual: number;
   123→   123→}
   124→   124→
   125→   125→function generateAccuracyData(symbol: string, currentPrice: number): AccuracyPoint[] {
   126→   126→  const rng = seededRandom(symbolSeed(symbol) + 999);
   127→   127→  const data: AccuracyPoint[] = [];
   128→   128→  let price = currentPrice * 0.92;
   129→   129→  const now = new Date();
   130→   130→
   131→   131→  for (let i = 30; i >= 0; i--) {
   132→   132→    const date = new Date(now);
   133→   133→    date.setDate(date.getDate() - i);
   134→   134→    if (date.getDay() === 0 || date.getDay() === 6) continue;
   135→   135→
   136→   136→    const change = (rng() - 0.47) * 0.03;
   137→   137→    const actual = parseFloat((price * (1 + change)).toFixed(2));
   138→   138→    const noise = (rng() - 0.5) * currentPrice * 0.015;
   139→   139→    const predicted = parseFloat((actual + noise).toFixed(2));
   140→   140→
   141→   141→    data.push({
   142→   142→      date: date.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   143→   143→      predicted,
   144→   144→      actual,
   145→   145→    });
   146→   146→    price = actual;
   147→   147→  }
   148→   148→  return data;
   149→   149→}
