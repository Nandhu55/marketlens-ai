   106→// ─── Chart Tooltip ───────────────────────────────────────────────────
   107→function ChartTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ value: number; dataKey: string; color?: string }>; label?: string }) {
   108→  if (!active || !payload?.length) return null;
   109→  return (
   110→    <div className="market-tooltip px-3 py-2 text-xs">
   111→      <p className="text-slate-400 mb-1">{label}</p>
   112→      {payload.map((entry, i) => (
   113→        <p key={i} style={{ color: entry.color || '#fff' }} className="font-semibold">
   114→          {entry.dataKey}: {typeof entry.value === 'number' ? entry.value.toFixed(2) : entry.value}
   115→        </p>
   116→      ))}
   117→    </div>
   118→  );
   119→}

   129→// ─── Main LiveMarketPage ────────────────────────────────────────────
   130→export function LiveMarketPage() {
   131→  const { selectedStock, setSelectedStock, liveTicks, wsConnectionStatus, staleSymbols, subscribedWsSymbols } = useAppStore();
   132→
   133→  // Initialize live WebSocket connection
   134→  useLiveMarket();
   135→
   136→  // Get live tick for selected stock
   137→  const selectedSymbol = selectedStock.symbol.toUpperCase();
   138→  const liveTick = liveTicks[selectedSymbol];
   139→  const isTickLive = liveTick && liveTick.dataStatus === 'LIVE' && !staleSymbols.includes(selectedSymbol);
   140→  const dataStatus = getDataStatus(selectedStock.symbol);
   141→  const [searchQuery, setSearchQuery] = useState('');
   142→  const [searchResults, setSearchResults] = useState<SearchResult[]>([]);
   143→  const [showSearch, setShowSearch] = useState(false);
   144→  const [activeTab, setActiveTab] = useState('price');
   145→  const [activeTimeframe, setActiveTimeframe] = useState(90);
   146→  const [stockData, setStockData] = useState<StockResponse | null>(null);
   147→  const [loading, setLoading] = useState(true);
   148→  const [searchLoading, setSearchLoading] = useState(false);
   149→  const searchRef = useRef<HTMLDivElement>(null);
   150→  const searchTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   151→
   152→  // Fetch stock data
   153→  const fetchStockData = useCallback(async (symbol: string, days: number) => {
   154→    setLoading(true);
   155→    try {
   156→      const res = await fetch(`/api/stock?symbol=${symbol}&days=${days}`);
   157→      const data: StockResponse = await res.json();
   158→      setStockData(data);
   159→    } catch {
   160→      setStockData(null);
   161→    } finally {
   162→      setLoading(false);
   163→    }
   164→  }, []);
   165→
   166→  // Initial fetch
   167→  useEffect(() => {
   168→    fetchStockData(selectedStock.symbol, activeTimeframe);
   169→  }, [selectedStock.symbol, activeTimeframe, fetchStockData]);
   170→
   171→  // Search with debounce
   172→  useEffect(() => {
   173→    if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current);
   174→    if (searchQuery.length < 1) {
   175→      setSearchResults([]);
   176→      return;
   177→    }
   178→    setSearchLoading(true);
   179→    searchTimeoutRef.current = setTimeout(async () => {
   180→      try {
   181→        const res = await fetch(`/api/search?q=${encodeURIComponent(searchQuery)}`);
   182→        const data = await res.json();
   183→        setSearchResults(data.results || []);
   184→      } catch {
   185→        setSearchResults([]);
   186→      } finally {
   187→        setSearchLoading(false);
   188→      }
   189→    }, 300);
   190→    return () => {
   191→      if (searchTimeoutRef.current) clearTimeout(searchTimeoutRef.current);
   192→    };
   193→  }, [searchQuery]);
   194→
   195→  // Close search on outside click
   196→  useEffect(() => {
   197→    const handleClick = (e: MouseEvent) => {
   198→      if (searchRef.current && !searchRef.current.contains(e.target as Node)) {
   199→        setShowSearch(false);
   200→      }
   201→    };
   202→    document.addEventListener('mousedown', handleClick);
   203→    return () => document.removeEventListener('mousedown', handleClick);
   204→  }, []);
   205→
   206→  const handleSelectStock = (result: SearchResult) => {
   207→    setSelectedStock({
   208→      symbol: result.symbol,
   209→      name: result.name,
   210→      price: result.price || 0,
   211→      change: result.change || 0,
   212→      changePercent: result.changePercent || 0,
   213→      volume: 0,
   214→      marketCap: 0,
   215→    });
   216→    setShowSearch(false);
   217→    setSearchQuery('');
   218→    setSearchResults([]);
   219→  };
   220→
   221→  // Prepare chart data
   222→  const ohlcv = stockData?.ohlcv || [];
   223→  const hasOHLCV = stockData?.hasOHLCV && ohlcv.length > 0;
   224→
   225→  const priceData = ohlcv.map((d) => ({
   226→    date: d.date.slice(5),
   227→    open: d.open,
   228→    high: d.high,
   229→    low: d.low,
   230→    close: d.close,
   231→    volume: d.volume,
   232→  }));
   233→
   234→  // Volume chart data
   235→  const volumeData = ohlcv.map((d) => ({
   236→    date: d.date.slice(5),
   237→    volume: d.volume,
   238→    close: d.close,
   239→    prevClose: d.open,
   240→  }));
   241→
   242→  // Indicator chart data from API series
   243→  const series = stockData?.series || null;
   244→  const indicatorData = series
   245→    ? ohlcv.map((d, i) => {
   246→        const point: Record<string, string | number> = { date: d.date.slice(5), close: d.close };
   247→        if (series.sma20?.[i] !== undefined && !isNaN(series.sma20[i])) point.sma20 = series.sma20[i];
   248→        if (series.sma50?.[i] !== undefined && !isNaN(series.sma50[i])) point.sma50 = series.sma50[i];
   249→        if (series.ema12?.[i] !== undefined && !isNaN(series.ema12[i])) point.ema12 = series.ema12[i];
   250→        if (series.rsi?.[i] !== undefined && !isNaN(series.rsi[i])) point.rsi = series.rsi[i];
   251→        if (series.bbUpper?.[i] !== undefined && !isNaN(series.bbUpper[i])) point.bbUpper = series.bbUpper[i];
   252→        if (series.bbMiddle?.[i] !== undefined && !isNaN(series.bbMiddle[i])) point.bbMiddle = series.bbMiddle[i];
   253→        if (series.bbLower?.[i] !== undefined && !isNaN(series.bbLower[i])) point.bbLower = series.bbLower[i];
   254→        // MACD series may be shorter — align from end
   255→        const macdLen = series.macd?.length || 0;
   256→        const macdIdx = i - (ohlcv.length - macdLen);
   257→        if (macdIdx >= 0 && series.macd?.[macdIdx] !== undefined && !isNaN(series.macd[macdIdx])) {
   258→          point.macd = series.macd[macdIdx];
   259→        }
   260→        const sigLen = series.macdSignal?.length || 0;
   261→        const sigIdx = i - (ohlcv.length - sigLen);
   262→        if (sigIdx >= 0 && series.macdSignal?.[sigIdx] !== undefined && !isNaN(series.macdSignal[sigIdx])) {
   263→          point.macdSignal = series.macdSignal[sigIdx];
   264→        }
   265→        const histLen = series.macdHistogram?.length || 0;
   266→        const histIdx = i - (ohlcv.length - histLen);
   267→        if (histIdx >= 0 && series.macdHistogram?.[histIdx] !== undefined && !isNaN(series.macdHistogram[histIdx])) {
   268→          point.macdHistogram = series.macdHistogram[histIdx];
   269→        }
   270→        return point;
   271→      })
   272→    : [];
   273→
   274→  const indicators = stockData?.indicators || null;
   275→
   276→  return (
   277→    <motion.div
   278→      variants={containerVariants}
   279→      initial="hidden"
   280→      animate="show"
   281→      className="space-y-6"
   282→    >
   283→      {/* ── Search Bar ────────────────────────────────────────── */}
   284→      <motion.div variants={itemVariants} ref={searchRef} className="relative">
   285→        <div className="relative">
   286→          <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   287→          <Input
   288→            value={searchQuery}
   289→            onChange={(e) => {
   290→              setSearchQuery(e.target.value);
   291→              setShowSearch(true);
   292→            }}
   293→            onFocus={() => setShowSearch(true)}
   294→            placeholder="Search stocks by name or symbol..."
   295→            className="glass-card pl-11 pr-10 py-3 h-12 text-sm bg-transparent border-white/10 focus:border-indigo-500/50 rounded-xl"
   296→          />
   297→          {searchQuery && (
   298→            <button
   299→              onClick={() => { setSearchQuery(''); setSearchResults([]); }}
   300→              className="absolute right-4 top-1/2 -translate-y-1/2 text-slate-500 hover:text-white transition-colors"
   301→            >
   302→              <X className="w-4 h-4" />
   303→            </button>
   304→          )}
   305→        </div>
   306→
   307→        {/* Search Dropdown */}
   308→        <AnimatePresence>
   309→          {showSearch && searchQuery.length >= 1 && (
   310→            <motion.div
   311→              initial={{ opacity: 0, y: -4 }}
   312→              animate={{ opacity: 1, y: 0 }}
   313→              exit={{ opacity: 0, y: -4 }}
   314→              transition={{ duration: 0.15 }}
   315→              className="absolute top-full left-0 right-0 mt-2 glass-card p-2 z-50 max-h-72 overflow-y-auto"
   316→            >
   317→              {searchLoading ? (
   318→                <div className="flex items-center justify-center py-6">
   319→                  <LoadingSpinner size={24} label="Searching..." />
   320→                </div>
   321→              ) : searchResults.length === 0 ? (
   322→                <div className="text-center py-6 text-slate-500 text-sm">No results found</div>
   323→              ) : (
   324→                searchResults.map((r) => (
   325→                  <button
   326→                    key={r.symbol}
   327→                    onClick={() => handleSelectStock(r)}
   328→                    className="w-full flex items-center justify-between px-3 py-2.5 rounded-xl hover:bg-white/5 transition-colors text-left"
   329→                  >
   330→                    <div>
   331→                      <div className="flex items-center gap-2">
   332→                        <span className="text-sm font-bold text-white">{r.symbol}</span>
   333→                        <Badge variant="outline" className="text-[9px] border-white/10 text-slate-400 px-1.5 py-0">
   334→                          {r.exchange}
   335→                        </Badge>
   336→                      </div>
   337→                      <p className="text-xs text-slate-400 mt-0.5">{r.name}</p>
   338→                    </div>
   339→                    <div className="text-right">
   340→                      {r.hasRealPrice ? (
   341→                        <>
   342→                          <p className="text-sm font-semibold text-white">${formatPrice(r.price)}</p>
   343→                          <p className={cn('text-xs font-medium', getChangeColor(r.changePercent))}>
   344→                            {formatPercent(r.changePercent)}
   345→                          </p>
   346→                        </>
   347→                      ) : (
   348→                        <p className="text-xs text-slate-500">No price</p>
   349→                      )}
   350→                    </div>
   351→                  </button>
   352→                ))
   353→              )}
   354→            </motion.div>
   355→          )}
   356→        </AnimatePresence>
   357→      </motion.div>
   358→
   359→      {/* ── Main Content ───────────────────────────────────────── */}
   360→      <div className="grid grid-cols-1 lg:grid-cols-4 gap-6">
   361→        {/* Chart Area */}
   362→        <motion.div variants={itemVariants} className="lg:col-span-3 space-y-4">
   363→          {/* Header with timeframe selector */}
   364→          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   365→            <div>
   366→              <div className="flex items-center gap-3">
   367→                <h2 className="text-xl font-bold text-white">{selectedStock.symbol}</h2>
   368→                {stockData?.stock?.name && (
   369→                  <span className="text-sm text-slate-400">{stockData.stock.name}</span>
   370→                )}
   371→                {isTickLive ? (
   372→                  <Badge className="bg-emerald-500/15 text-emerald-400 text-[10px] font-semibold border-0">
   373→                    <span className="inline-block w-1.5 h-1.5 rounded-full bg-emerald-500 mr-1.5 animate-pulse" />
   374→                    LIVE
   375→                  </Badge>
   376→                ) : dataStatus === 'CONNECTING' ? (
   377→                  <Badge className="bg-amber-500/15 text-amber-400 text-[10px] font-semibold border-0">
   378→                    CONNECTING
   379→                  </Badge>
   380→                ) : stockData?.realQuote && (
   381→                  <Badge className="bg-slate-500/15 text-slate-400 text-[10px] font-semibold border-0">
   382→                    REST
   383→                  </Badge>
   384→                )}
   385→              </div>
   386→              {(stockData?.stock || isTickLive) && (
   387→                <div className="flex items-center gap-3 mt-1">
   388→                  <span className="text-2xl font-bold text-white">
   389→                    {isTickLive
   390→                      ? (isIndianSymbol(selectedStock.symbol) ? `₹${formatPrice(liveTick.price)}` : `$${formatPrice(liveTick.price)}`)
   391→                      : (isIndianSymbol(selectedStock.symbol) ? `₹${formatPrice(stockData!.stock!.price)}` : `$${formatPrice(stockData!.stock!.price)}`)
   392→                    }
   393→                  </span>
   394→                  <span className={cn('text-sm font-semibold flex items-center gap-1', getChangeColor(isTickLive ? liveTick.changePercent : stockData!.stock!.changePercent))}>
   395→                    {(isTickLive ? liveTick.changePercent : stockData!.stock!.changePercent) >= 0 ? <ArrowUpRight className="w-4 h-4" /> : <ArrowDownRight className="w-4 h-4" />}
   396→                    {formatPrice(isTickLive ? liveTick.change : stockData!.stock!.change)} ({formatPercent(isTickLive ? liveTick.changePercent : stockData!.stock!.changePercent)})
   397→                  </span>
   398→                </div>
   399→              )}
   400→            </div>
   401→            <div className="flex items-center gap-2">
   402→              <div className="flex bg-white/5 rounded-xl p-1">
   403→                {TIMEFRAMES.map((tf) => (
   404→                  <button
   405→                    key={tf.days}
   406→                    onClick={() => setActiveTimeframe(tf.days)}
   407→                    className={cn(
   408→                      'px-3 py-1.5 text-xs font-medium rounded-lg transition-all',
   409→                      activeTimeframe === tf.days
   410→                        ? 'gradient-button text-white'
   411→                        : 'text-slate-400 hover:text-white hover:bg-white/5'
   412→                    )}
   413→                  >
   414→                    {tf.label}
   415→                  </button>
   416→                ))}
   417→              </div>
   418→            </div>
   419→          </div>
   420→
   421→          {/* Indicator Tabs + Chart */}
   422→          <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   423→            <TabsList className="bg-white/5 rounded-xl p-1 h-9">
   424→              <TabsTrigger value="price" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">Price</TabsTrigger>
   425→              <TabsTrigger value="sma" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">SMA/EMA</TabsTrigger>
   426→              <TabsTrigger value="rsi" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">RSI</TabsTrigger>
   427→              <TabsTrigger value="macd" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">MACD</TabsTrigger>
   428→              <TabsTrigger value="bb" className="text-xs data-[state=active]:gradient-button rounded-lg h-7">Bollinger</TabsTrigger>
   429→            </TabsList>
   430→
   431→            <TabsContent value="price" className="mt-4">
   432→              {loading ? (
   433→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   434→                  <LoadingSpinner label="Loading price data..." />
   435→                </div>
   436→              ) : !hasOHLCV ? (
   437→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   438→                  <EmptyState
   439→                    icon={<Activity className="w-8 h-8" />}
   440→                    title="No Live Data Available"
   441→                    description={`No OHLCV data for ${selectedStock.symbol}. Try a different stock or refresh.`}
   442→                    action={
   443→                      <Button variant="outline" size="sm" onClick={() => fetchStockData(selectedStock.symbol, activeTimeframe)} className="gradient-button border-0">
   444→                        <RefreshCw className="w-3.5 h-3.5 mr-2" />
   445→                        Retry
   446→                      </Button>
   447→                    }
   448→                  />
   449→                </div>
   450→              ) : (
   451→                <>
   452→                  <div className="glass-card p-4">
   453→                    <ResponsiveContainer width="100%" height={300}>
   454→                      <ComposedChart data={priceData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>
   455→                        <defs>
   456→                          <linearGradient id="areaGrad" x1="0" y1="0" x2="0" y2="1">
   457→                            <stop offset="0%" stopColor="#6366f1" stopOpacity={0.15} />
   458→                            <stop offset="100%" stopColor="#6366f1" stopOpacity={0} />
   459→                          </linearGradient>
   460→                        </defs>
   461→                        <CartesianGrid strokeDasharray="3 3" stroke="rgba(148,163,184,0.06)" />
   462→                        <XAxis dataKey="date" axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} />
   463→                        <YAxis domain={['auto', 'auto']} axisLine={false} tickLine={false} tick={{ fill: '#64748b', fontSize: 10 }} tickFormatter={(v: number) => `$${v.toFixed(0)}`} />
   464→                        <Tooltip content={<ChartTooltip />} />
   465→                        <Area type="monotone" dataKey="close" stroke="#6366f1" strokeWidth={2} fill="url(#areaGrad)" dot={false} />
   466→                        <Line type="monotone" dataKey="high" stroke="#10b981" strokeWidth={0.5} dot={false} strokeDasharray="2 2" />
   467→                        <Line type="monotone" dataKey="low" stroke="#ef4444" strokeWidth={0.5} dot={false} strokeDasharray="2 2" />
   468→                      </ComposedChart>
   469→                    </ResponsiveContainer>
   470→                  </div>
   471→                  {/* Volume bars */}
   472→                  <div className="glass-card p-4 mt-2">
   473→                    <div className="flex items-center gap-2 mb-2">
   474→                      <Volume2 className="w-3.5 h-3.5 text-slate-500" />
   475→                      <span className="text-xs text-slate-500 uppercase tracking-wider font-medium">Volume</span>
   476→                    </div>
   477→                    <ResponsiveContainer width="100%" height={80}>
   478→                      <BarChart data={volumeData} margin={{ top: 0, right: 5, left: -20, bottom: 0 }}>
   479→                        <XAxis dataKey="date" axisLine={false} tickLine={false} tick={false} />
   480→                        <YAxis axisLine={false} tickLine={false} tick={false} />
   481→                        <Tooltip content={<ChartTooltip />} />
   482→                        <Bar dataKey="volume" radius={[2, 2, 0, 0]}>
   483→                          {volumeData.map((entry, index) => (
   484→                            <Cell
   485→                              key={index}
   486→                              fill={entry.close >= entry.prevClose ? 'rgba(16,185,129,0.4)' : 'rgba(239,68,68,0.4)'}
   487→                            />
   488→                          ))}
   489→                        </Bar>
   490→                      </BarChart>
   491→                    </ResponsiveContainer>
   492→                  </div>
   493→                </>
   494→              )}
   495→            </TabsContent>
   496→
   497→            <TabsContent value="sma" className="mt-4">
   498→              {loading || !hasOHLCV || !series ? (
   499→                <div className="glass-card p-4 h-96 flex items-center justify-center">
   500→                  {!hasOHLCV ? (
   501→                    <EmptyState icon={<Layers className="w-8 h-8" />} title="No Live Data Available" description="No OHLCV data to compute indicators" />
   502→                  ) : (
   503→                    <LoadingSpinner label="Loading indicators..." />
   504→                  )}
