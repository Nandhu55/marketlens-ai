   113→   113→// ─── Fear & Greed Semicircle Gauge ───────────────────────────────────
   114→   114→function FearGreedGauge({ value }: { value: number }) {
   115→   115→  if (value <= 0) {
   116→   116→    return (
   117→   117→      <div className="flex flex-col items-center justify-center h-full">
   118→   118→        <p className="text-slate-500 text-sm">No Data</p>
   119→   119→      </div>
   120→   120→    );
   121→   121→  }
   122→   122→
   123→   123→  const clamped = Math.max(0, Math.min(100, value));
   124→   124→
   125→   125→  const getZone = () => {
   126→   126→    if (clamped <= 25) return { label: 'Extreme Fear', color: '#ef4444' };
   127→   127→    if (clamped <= 45) return { label: 'Fear', color: '#f97316' };
   128→   128→    if (clamped <= 55) return { label: 'Neutral', color: '#eab308' };
   129→   129→    if (clamped <= 75) return { label: 'Greed', color: '#22c55e' };
   130→   130→    return { label: 'Extreme Greed', color: '#10b981' };
   131→   131→  };
   132→   132→
   133→   133→  const zone = getZone();
   134→   134→  const width = 260;
   135→   135→  const height = 145;
   136→   136→  const cx = width / 2;
   137→   137→  const cy = height - 10;
   138→   138→  const radius = 100;
   139→   139→  const strokeWidth = 18;
   140→   140→  const valueAngle = Math.PI - (clamped / 100) * Math.PI;
   141→   141→  const needleLen = radius - strokeWidth - 5;
   142→   142→  const needleTipX = cx - needleLen * Math.cos(valueAngle);
   143→   143→  const needleTipY = cy - needleLen * Math.sin(valueAngle);
   144→   144→  const startX = cx - radius;
   145→   145→  const endX = cx + radius;
   146→   146→
   147→   147→  const segments = [
   148→   148→    { from: 0, to: 25, color: '#ef4444' },
   149→   149→    { from: 25, to: 45, color: '#f97316' },
   150→   150→    { from: 45, to: 55, color: '#eab308' },
   151→   151→    { from: 55, to: 75, color: '#22c55e' },
   152→   152→    { from: 75, to: 100, color: '#10b981' },
   153→   153→  ];
   154→   154→
   155→   155→  const getArcPath = (fromPct: number, toPct: number) => {
   156→   156→    const startAngle = Math.PI - (fromPct / 100) * Math.PI;
   157→   157→    const endAngle = Math.PI - (toPct / 100) * Math.PI;
   158→   158→    const sx = cx + radius * Math.cos(startAngle);
   159→   159→    const sy = cy - radius * Math.sin(startAngle);
   160→   160→    const ex = cx + radius * Math.cos(endAngle);
   161→   161→    const ey = cy - radius * Math.sin(endAngle);
   162→   162→    const largeArc = (fromPct - toPct) > 50 ? 1 : 0;
   163→   163→    return `M ${sx} ${sy} A ${radius} ${radius} 0 ${largeArc} 0 ${ex} ${ey}`;
   164→   164→  };
   165→   165→
   166→   166→  return (
   167→   167→    <div className="flex flex-col items-center">
   168→   168→      <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`}>
   169→   169→        {segments.map((seg) => (
   170→   170→          <path
   171→   171→            key={`${seg.from}-${seg.to}`}
   172→   172→            d={getArcPath(seg.from, seg.to)}
   173→   173→            fill="none"
   174→   174→            stroke={seg.color}
   175→   175→            strokeWidth={strokeWidth}
   176→   176→            strokeLinecap="butt"
   177→   177→            opacity={0.2}
   178→   178→          />
   179→   179→        ))}
   180→   180→        <motion.path
   181→   181→          d={getArcPath(0, clamped)}
   182→   182→          fill="none"
   183→   183→          stroke={zone.color}
   184→   184→          strokeWidth={strokeWidth}
   185→   185→          strokeLinecap="round"
   186→   186→          initial={{ pathLength: 0 }}
   187→   187→          animate={{ pathLength: 1 }}
   188→   188→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   189→   189→          style={{ filter: `drop-shadow(0 0 8px ${zone.color}60)` }}
   190→   190→        />
   191→   191→        <motion.line
   192→   192→          x1={cx} y1={cy} x2={needleTipX} y2={needleTipY}
   193→   193→          stroke={zone.color} strokeWidth={2.5} strokeLinecap="round"
   194→   194→          initial={{ opacity: 0 }}
   195→   195→          animate={{ opacity: 1, x2: needleTipX, y2: needleTipY }}
   196→   196→          transition={{ duration: 1, delay: 0.5 }}
   197→   197→          style={{ filter: `drop-shadow(0 0 4px ${zone.color}80)` }}
   198→   198→        />
   199→   199→        <circle cx={cx} cy={cy} r={5} fill={zone.color} />
   200→   200→        <text x={startX - 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">0</text>
   201→   201→        <text x={endX + 5} y={cy + 14} textAnchor="middle" className="fill-slate-500" fontSize="9">100</text>
   202→   202→      </svg>
   203→   203→      <div className="text-center -mt-1">
   204→   204→        <div className="text-3xl font-bold" style={{ color: zone.color }}>
   205→   205→          {Math.round(clamped)}
   206→   206→        </div>
   207→   207→        <div className="text-xs font-medium mt-0.5" style={{ color: zone.color }}>
   208→   208→          {zone.label}
   209→   209→        </div>
   210→   210→        <div className="text-[10px] text-slate-500 mt-1 uppercase tracking-widest">Fear &amp; Greed Index</div>
   211→   211→      </div>
   212→   212→    </div>
   213→   213→  );
   214→   214→}

   269→   269→      setLoading(false);

   277→   277→      const data: StockDataResponse = await res.json();

   280→   280→      setStockData(null);

   296→   296→    symbol,

   316→   316→      })()
   317→   317→    : 0;
   318→   318→
   319→   319→  const topGainer = overviewData?.overview?.topGainers?.[0];
   320→   320→  const topLoser = overviewData?.overview?.topLosers?.[0];
